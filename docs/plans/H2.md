## Global Constraints

- Python `>=3.11,<3.13`; Görev 4'te `<3.14`'e genişler (Colab T4 = 3.13, Spike C), CI 3.11 + 3.13. Paket `baykus`, CLI adları `baykus-*`. Python komutları `C:\baykus\python` içinden `uv run ...`.
- CI `uv sync --frozen` ile **extra'sız** kurar: torch/ultralytics/mlflow import eden testler `pytest.importorskip` kullanır; uçtan uca testler `@pytest.mark.slow` (CI'da çalışmaz). Ultralytics/torch/mlflow modül seviyesinde import edilmez.
- `data/`, `runs/`, `mlruns/`, `*.pt`, `*.onnx`, `*.engine` asla commit edilmez. Tahmin dosyaları (`*.json.gz`) `runs/` altında kalır.
- Metin dosyaları `encoding="utf-8"`. Görüntü I/O `baykus.io.imread_any` / `imwrite_any`.
- Sınıf sırası: `car, truck, bus, van, freight_car`; COCO `category_id = cls_id + 1`.
- pycocotools'ta `maxDets` en az **1000** (`coco_eval.MAX_DETS`). Ultralytics val'de `max_det=1000`, `conf=0.001`.
- Ana etiket seti **IR** (ADR 0003). Modlar: `rgb`, `ir`, `early_fusion`; modalite klasörleri `rgb`, `ir`, `rgbt4`.
- Test seti sadece final ölçümde. `baykus-eval` varsayılan split `val`; `test` açıkça istenir.
- Varsayılan hiperparametreler spec §10.1: imgsz 640, epochs 100, patience 20, batch 16, seed 42 + `deterministic=True`, close_mosaic 10, cache False, amp True, save_period 5, workers 4.
- Bootstrap: görüntü düzeyinde, 1000 yeniden örnekleme, %95 yüzdelik aralık; eşleştirilmiş farkta aralık 0'ı içermiyorsa "anlamlı".
- Sonuç JSON'larında ölçülmemiş değer `null`; tahmini değer asla yazılmaz.
- Token/şifre hiçbir dosyaya yazılmaz. Conventional Commits, atomik commit, Claude imzası yok.

```python
"""Deney config şeması (spec §10.3). Bilinmeyen anahtar her seviyede ConfigError verir (FR-T2)."""

from __future__ import annotations

import hashlib
import json
from dataclasses import dataclass, field, fields
from pathlib import Path

import yaml

MODES = ("rgb", "ir", "early_fusion")
MODALITY_DIR = {"rgb": "rgb", "ir": "ir", "early_fusion": "rgbt4"}
LABEL_SETS = ("ir", "rgb")
SPLITS = ("train", "val", "test")
SLICES = ("illumination", "size", "class")
FIRST_CONV_INITS = ("ul_default", "rgb_mean")
_TOP = {
    "experiment", "mode", "model", "data", "label_set", "first_conv_init",
    "modality_dropout", "train", "eval", "tracking",
}


class ConfigError(ValueError):
    pass


@dataclass(frozen=True)
class TrainCfg:
    imgsz: int = 640
    epochs: int = 100
    patience: int = 20
    batch: int = 16
    seed: int = 42
    close_mosaic: int = 10
    workers: int = 4
    save_period: int = 5


@dataclass(frozen=True)
class EvalCfg:
    splits: tuple[str, ...] = ("val",)
    label_sets: tuple[str, ...] = LABEL_SETS
    slices: tuple[str, ...] = SLICES
    n_boot: int = 1000


@dataclass(frozen=True)
class DropoutCfg:
    p_rgb: float = 0.0
    p_ir: float = 0.0


@dataclass(frozen=True)
class ExperimentConfig:
    experiment: str
    mode: str
    model: str
    data: str
    label_set: str = "ir"
    first_conv_init: str | None = None
    modality_dropout: DropoutCfg | None = None
    train: TrainCfg = field(default_factory=TrainCfg)
    eval: EvalCfg = field(default_factory=EvalCfg)
    mlflow_experiment: str = "baykus"
    config_hash: str = ""

    @property
    def modality(self) -> str:
        return MODALITY_DIR[self.mode]


def config_hash(raw: dict) -> str:
    blob = json.dumps(raw, sort_keys=True, ensure_ascii=False, separators=(",", ":"))
    return "sha256:" + hashlib.sha256(blob.encode("utf-8")).hexdigest()


def _check_keys(d: object, allowed: set[str], where: str) -> dict:
    if not isinstance(d, dict):
        raise ConfigError(f"{where}: sözlük bekleniyordu")
    extra = sorted(set(d) - allowed)
    if extra:
        raise ConfigError(f"{where}: bilinmeyen anahtar(lar): {', '.join(extra)}")
    return d


def _section(cls: type, d: object, where: str):
    d = _check_keys(d if d is not None else {}, {f.name for f in fields(cls)}, where)
    return cls(**{k: tuple(v) if isinstance(v, list) else v for k, v in d.items()})


def _check_subset(values: tuple[str, ...], allowed: tuple[str, ...], where: str) -> None:
    bad = [v for v in values if v not in allowed]
    if bad:
        raise ConfigError(f"{where}: geçersiz değer(ler) {bad}; seçenekler: {allowed}")


def parse_config(raw: dict) -> ExperimentConfig:
    _check_keys(raw, _TOP, "config")
    for key in ("experiment", "mode", "model", "data"):
        if key not in raw:
            raise ConfigError(f"config: zorunlu anahtar eksik: {key}")
    mode = raw["mode"]
    if mode not in MODES:
        raise ConfigError(f"mode: {mode!r} geçersiz; seçenekler: {MODES}")
    label_set = raw.get("label_set", "ir")
    if label_set not in LABEL_SETS:
        raise ConfigError(f"label_set: {label_set!r} geçersiz; seçenekler: {LABEL_SETS}")
    expected = f"data_{MODALITY_DIR[mode]}.yaml"
    if Path(str(raw["data"])).name != expected:
        raise ConfigError(f"data: mode={mode} için {expected} bekleniyordu, {raw['data']!r} verildi")

    init, drop_raw = raw.get("first_conv_init"), raw.get("modality_dropout")
    if mode != "early_fusion" and (init is not None or drop_raw is not None):
        raise ConfigError("first_conv_init ve modality_dropout sadece mode=early_fusion için")
    if mode == "early_fusion":
        init = init or "ul_default"
        if init not in FIRST_CONV_INITS:
            raise ConfigError(
                f"first_conv_init: {init!r} geçersiz; seçenekler: {FIRST_CONV_INITS} "
                "('random' H4'te, bütçe varsa)"
            )
    drop = None
    if drop_raw is not None:
        drop = _section(DropoutCfg, drop_raw, "modality_dropout")
        for name in ("p_rgb", "p_ir"):
            p = getattr(drop, name)
            if not 0.0 <= p <= 1.0:
                raise ConfigError(f"modality_dropout.{name}: 0-1 arası olmalı, {p} verildi")
        if drop.p_rgb + drop.p_ir > 1.0:
            raise ConfigError(
                "modality_dropout: p_rgb + p_ir en fazla 1 (aynı örnekte ikisi birden sıfırlanmaz)"
            )

    train = _section(TrainCfg, raw.get("train"), "train")
    ev = _section(EvalCfg, raw.get("eval"), "eval")
    _check_subset(ev.splits, SPLITS, "eval.splits")
    _check_subset(ev.label_sets, LABEL_SETS, "eval.label_sets")
    _check_subset(ev.slices, SLICES, "eval.slices")
    tracking = _check_keys(raw.get("tracking") or {}, {"mlflow_experiment"}, "tracking")

    return ExperimentConfig(
        experiment=str(raw["experiment"]),
        mode=mode,
        model=str(raw["model"]),
        data=str(raw["data"]),
        label_set=label_set,
        first_conv_init=init,
        modality_dropout=drop,
        train=train,
        eval=ev,
        mlflow_experiment=str(tracking.get("mlflow_experiment", "baykus")),
        config_hash=config_hash(raw),
    )


def load_config(path: Path) -> ExperimentConfig:
    raw = yaml.safe_load(Path(path).read_text(encoding="utf-8"))
    if not isinstance(raw, dict):
        raise ConfigError(f"{path}: config bir sözlük (YAML mapping) olmalı")
    return parse_config(raw)
```

```python
"""Çalıştırma kimliği ve izlenebilirlik: git commit, run_id, JSON yazma (FR-T5, S1)."""

from __future__ import annotations

import datetime as dt
import json
import platform
import re
import subprocess
from importlib.metadata import PackageNotFoundError, version
from pathlib import Path

SOFTWARE = ("torch", "ultralytics", "numpy", "pycocotools")
_MODEL_RE = re.compile(r"(yolo\d+)([nsmlx])")


def git_info(cwd: Path | None = None) -> dict:
    def run(*args: str) -> str:
        return subprocess.run(
            ["git", *args], cwd=cwd, capture_output=True, text=True, check=True
        ).stdout.strip()

    try:
        commit = run("rev-parse", "HEAD")
        dirty = bool(run("status", "--porcelain", "--untracked-files=no"))
    except (OSError, subprocess.CalledProcessError):
        return {"git_commit": None, "git_dirty": None}
    return {"git_commit": commit, "git_dirty": dirty}


def _parse_model(model: str) -> re.Match:
    m = _MODEL_RE.fullmatch(Path(model).stem)
    if m is None:
        raise ValueError(f"model adı çözümlenemedi (ör. yolo26s.pt bekleniyordu): {model}")
    return m


def model_family(model: str) -> str:
    return _parse_model(model).group(1)


def model_size(model: str) -> str:
    return _parse_model(model).group(2)


def make_run_id(experiment: str, model: str, commit: str | None, when: dt.datetime) -> str:
    short = (commit or "nogit")[:7]
    return f"{when:%Y-%m-%d}_{experiment.replace('_', '-')}_{model_size(model)}_{short}"


def utc_now() -> dt.datetime:
    return dt.datetime.now(dt.timezone.utc)


def iso(t: dt.datetime) -> str:
    return t.astimezone(dt.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def write_json(path: Path, data: dict) -> Path:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    return path


def software_versions() -> dict:
    out: dict[str, str | None] = {"python": platform.python_version()}
    for pkg in SOFTWARE:
        try:
            out[pkg] = version(pkg)
        except PackageNotFoundError:
            out[pkg] = None
    return out
```

```python
"""Erken füzyon eğitimi: modality dropout (FR-T7) ve rgb_mean başlatması (FR-T8).

Ultralytics'e yama yapılmaz: dropout DetectionTrainer.preprocess_batch alt sınıfında,
rgb_mean on_pretrain_routine_end callback'inde uygulanır (H1 Spike A). Validasyon etkilenmez.
"""

from __future__ import annotations

from collections.abc import Callable

import numpy as np

from baykus.train.config import ExperimentConfig

KEEP, DROP_RGB, DROP_IR = 0, 1, 2


def sample_drop_codes(n: int, p_rgb: float, p_ir: float, rng: np.random.Generator) -> np.ndarray:
    """Örnek başına kod: KEEP, DROP_RGB veya DROP_IR. Aynı örnekte ikisi birden asla."""
    u = rng.random(n)
    codes = np.full(n, KEEP, dtype=np.int8)
    codes[u < p_rgb] = DROP_RGB
    codes[(u >= p_rgb) & (u < p_rgb + p_ir)] = DROP_IR
    return codes


def apply_drop_codes(img, codes: np.ndarray):
    """img: torch (B, 4, H, W). Kanal 0-2 = RGB, kanal 3 = T (ADR 0004). Yerinde sıfırlar."""
    import torch

    c = torch.as_tensor(codes, device=img.device)
    img[c == DROP_RGB, :3] = 0
    img[c == DROP_IR, 3:] = 0
    return img


def set_rgb_mean(trainer) -> None:
    """Callback: ilk katmanın T kanalını RGB ağırlıklarının ortalamasıyla başlatır (model + EMA)."""
    import torch

    ema = getattr(trainer, "ema", None)
    with torch.no_grad():
        for m in (trainer.model, ema.ema if ema is not None else None):
            if m is None:
                continue
            w = m.model[0].conv.weight
            if w.shape[1] != 4:
                raise ValueError(f"rgb_mean: ilk katman 4 kanal bekleniyordu, {w.shape[1]} bulundu")
            w[:, 3] = w[:, :3].mean(1)
    trainer.rgb_mean_applied = True


def callbacks_for(cfg: ExperimentConfig, resume: bool) -> list[tuple[str, Callable]]:
    if cfg.first_conv_init == "rgb_mean" and not resume:
        return [("on_pretrain_routine_end", set_rgb_mean)]
    return []


def make_trainer_class(cfg: ExperimentConfig) -> type:
    from ultralytics.models.yolo.detect import DetectionTrainer

    drop = cfg.modality_dropout
    if drop is None or (drop.p_rgb == 0 and drop.p_ir == 0):
        return DetectionTrainer
    p_rgb, p_ir, seed = drop.p_rgb, drop.p_ir, cfg.train.seed

    class ModalityDropoutTrainer(DetectionTrainer):
        _rng = np.random.default_rng(seed)

        def preprocess_batch(self, batch: dict) -> dict:
            batch = super().preprocess_batch(batch)
            codes = sample_drop_codes(batch["img"].shape[0], p_rgb, p_ir, self._rng)
            apply_drop_codes(batch["img"], codes)
            return batch

    return ModalityDropoutTrainer
```

`python/src/baykus/eval/predictions.py`:
```python
"""Ultralytics predictions.json → bizim COCO image_id'lerimiz; tahmin dosyası (kanca K3).

Ultralytics image_id'yi dosya adından (int(stem)) üretir; bizim COCO kimliklerimiz sıralı
numaradır ve dışlanan dosyalar yüzünden kayar. Eşleme bu yüzden dosya adı üzerinden yapılır.
"""

from __future__ import annotations

import gzip
import json
from pathlib import Path

from pycocotools.coco import COCO


def remap_to_gt(preds: list[dict], gt: COCO) -> tuple[list[dict], int]:
    by_stem = {Path(im["file_name"]).stem: im["id"] for im in gt.dataset["images"]}
    by_int = {int(s): i for s, i in by_stem.items() if s.isdigit()}
    out, dropped = [], 0
    for p in preds:
        if "file_name" in p:
            img_id = by_stem.get(Path(p["file_name"]).stem)
        elif isinstance(p["image_id"], int):
            img_id = by_int.get(p["image_id"])
        else:
            img_id = by_stem.get(str(p["image_id"]))
        if img_id is None:
            dropped += 1
            continue
        out.append(
            {
                "image_id": img_id,
                "category_id": int(p["category_id"]),
                "bbox": [float(x) for x in p["bbox"]],
                "score": float(p["score"]),
            }
        )
    return out, dropped


def save_predictions(path: Path, preds: list[dict]) -> Path:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with gzip.open(path, "wt", encoding="utf-8") as f:
        json.dump(preds, f)
    return path


def load_predictions(path: Path) -> list[dict]:
    with gzip.open(Path(path), "rt", encoding="utf-8") as f:
        return json.load(f)
```

```python
"""Görüntü düzeyinde bootstrap güven aralığı ve eşleştirilmiş fark (FR-V8).

pycocotools evaluate() bir kez çalışır (IoU 0.5, alan 'all', maxDets=MAX_DETS); her yeniden
örneklemede accumulate adımları burada birebir tekrarlanır. Örneklem = kendisi olduğunda sonuç
evaluate_coco()['map50'] ile aynıdır (testli).
"""

from __future__ import annotations

import contextlib
import io
from dataclasses import dataclass

import numpy as np
from pycocotools.coco import COCO
from pycocotools.cocoeval import COCOeval

from baykus.eval.coco_eval import MAX_DETS

_REC_THRS = np.linspace(0.0, 1.0, 101)


@dataclass(frozen=True)
class ClassMatches:
    offsets: np.ndarray  # (I+1,) görüntü başına tespit dilimleri
    scores: np.ndarray  # (D,)
    tp: np.ndarray  # (D,) bool: IoU>=0.5 eşleşti ve yok sayılmadı
    fp: np.ndarray  # (D,) bool: eşleşmedi ve yok sayılmadı
    npig: np.ndarray  # (I,) yok sayılmayan GT sayısı


@dataclass(frozen=True)
class Matches:
    img_ids: list[int]
    classes: dict[str, ClassMatches]


def _pack(per_img: list[tuple[np.ndarray, np.ndarray, np.ndarray, int]]) -> ClassMatches:
    lens = np.array([len(s) for s, _, _, _ in per_img], dtype=np.int64)
    offsets = np.concatenate([[0], np.cumsum(lens)]).astype(np.int64)

    def cat(i: int, dtype) -> np.ndarray:
        parts = [np.asarray(x[i], dtype=dtype) for x in per_img]
        return np.concatenate(parts) if parts else np.zeros(0, dtype)

    return ClassMatches(
        offsets=offsets,
        scores=cat(0, np.float64),
        tp=cat(1, bool),
        fp=cat(2, bool),
        npig=np.array([x[3] for x in per_img], dtype=np.int64),
    )


def collect_matches(gt: COCO, dets: list[dict], img_ids: list[int] | None = None) -> Matches:
    ids = sorted(img_ids) if img_ids is not None else sorted(gt.getImgIds())
    id_set = set(ids)
    dets = [d for d in dets if d["image_id"] in id_set]
    cats = gt.loadCats(sorted(gt.getCatIds()))  # COCOeval p.catIds sırası; evalImgs indeksi buna bağlı
    evals = None
    if dets and ids:
        with contextlib.redirect_stdout(io.StringIO()):
            ev = COCOeval(gt, gt.loadRes(dets), iouType="bbox")
            ev.params.imgIds = ids
            ev.params.iouThrs = np.array([0.5])
            ev.params.maxDets = [MAX_DETS]
            ev.params.areaRng = [[0, 1e10]]
            ev.params.areaRngLbl = ["all"]
            ev.evaluate()
        evals = ev.evalImgs
    empty = (np.zeros(0), np.zeros(0, bool), np.zeros(0, bool))
    classes = {}
    for k, c in enumerate(cats):
        per_img = []
        for i, img in enumerate(ids):
            if evals is None:
                n_gt = len(gt.getAnnIds(imgIds=[img], catIds=[c["id"]], iscrowd=False))
                per_img.append((*empty, n_gt))
                continue
            e = evals[k * len(ids) + i]
            if e is None:
                per_img.append((*empty, 0))
                continue
            dtm = np.asarray(e["dtMatches"][0])
            dti = np.asarray(e["dtIgnore"][0]).astype(bool)
            gti = np.asarray(e["gtIgnore"])
            per_img.append(
                (np.asarray(e["dtScores"]), (dtm > 0) & ~dti, (dtm == 0) & ~dti,
                 int((gti == 0).sum()))
            )
        classes[c["name"]] = _pack(per_img)
    return Matches(ids, classes)


def _take(cm: ClassMatches, s: np.ndarray) -> ClassMatches:
    lens = np.diff(cm.offsets)[s]
    starts = cm.offsets[:-1][s]
    total = int(lens.sum())
    if total:
        shift = starts - np.concatenate([[0], np.cumsum(lens)[:-1]])
        idx = np.repeat(shift, lens) + np.arange(total)
    else:
        idx = np.zeros(0, dtype=np.int64)
    return ClassMatches(
        offsets=np.concatenate([[0], np.cumsum(lens)]).astype(np.int64),
        scores=cm.scores[idx],
        tp=cm.tp[idx],
        fp=cm.fp[idx],
        npig=cm.npig[s],
    )


def subset(m: Matches, img_ids: list[int]) -> Matches:
    pos = {img: i for i, img in enumerate(m.img_ids)}
    keep = sorted(i for i in set(img_ids) if i in pos)
    s = np.array([pos[i] for i in keep], dtype=np.int64)
    return Matches(keep, {name: _take(cm, s) for name, cm in m.classes.items()})


def _ap(cm: ClassMatches) -> float | None:
    npig = int(cm.npig.sum())
    if npig == 0:
        return None  # pycocotools: precision -1 → ortalamaya girmez
    order = np.argsort(-cm.scores, kind="mergesort")
    tp_sum = np.cumsum(cm.tp[order]).astype(float)
    fp_sum = np.cumsum(cm.fp[order]).astype(float)
    q = np.zeros(len(_REC_THRS))
    if len(tp_sum):
        rc = tp_sum / npig
        pr = tp_sum / (fp_sum + tp_sum + np.spacing(1))
        pr = np.maximum.accumulate(pr[::-1])[::-1]
        inds = np.searchsorted(rc, _REC_THRS, side="left")
        ok = inds < len(pr)
        q[ok] = pr[inds[ok]]
    return float(q.mean())


def ap50(m: Matches, sample: np.ndarray | None = None) -> float:
    s = np.arange(len(m.img_ids)) if sample is None else sample
    aps = [a for a in (_ap(_take(cm, s)) for cm in m.classes.values()) if a is not None]
    return float(np.mean(aps)) if aps else 0.0


def bootstrap_ci(m: Matches, n: int = 1000, seed: int = 0) -> tuple[float | None, float | None]:
    k = len(m.img_ids)
    if k == 0:
        return (None, None)
    rng = np.random.default_rng(seed)
    vals = np.array([ap50(m, rng.integers(0, k, k)) for _ in range(n)])
    lo, hi = np.percentile(vals, [2.5, 97.5])
    return float(lo), float(hi)


def paired_diff_ci(a: Matches, b: Matches, n: int = 1000, seed: int = 0) -> dict:
    if a.img_ids != b.img_ids:
        raise ValueError("eşleştirilmiş bootstrap için iki model aynı görüntü listesinde ölçülmeli")
    k = len(a.img_ids)
    rng = np.random.default_rng(seed)
    diffs = []
    for _ in range(n):
        s = rng.integers(0, k, k)
        diffs.append(ap50(a, s) - ap50(b, s))
    lo, hi = (float(x) for x in np.percentile(diffs, [2.5, 97.5]))
    return {"diff": ap50(a) - ap50(b), "ci95": [lo, hi], "significant": bool(lo > 0 or hi < 0)}


def precision_recall(m: Matches, conf: float = 0.25) -> tuple[float | None, float | None]:
    tp = fp = npig = 0
    for cm in m.classes.values():
        keep = cm.scores >= conf
        tp += int(cm.tp[keep].sum())
        fp += int(cm.fp[keep].sum())
        npig += int(cm.npig.sum())
    precision = tp / (tp + fp) if tp + fp else None
    recall = tp / npig if npig else None
    return precision, recall
```

"""Bir tahmin listesi için spec §12.4 metrik bloğu (FR-V1, V2, V8)."""

from __future__ import annotations

from pycocotools.coco import COCO

from baykus.eval.bootstrap import bootstrap_ci, collect_matches, precision_recall, subset
from baykus.eval.coco_eval import evaluate_coco

ILLUMINATION = ("day", "night", "dark_night")


def build_metrics(
    gt: COCO,
    dets: list[dict],
    illumination: dict[str, list[int]] | None,
    n_boot: int = 1000,
    seed: int = 0,
) -> dict:
    base = evaluate_coco(gt, dets)
    m = collect_matches(gt, dets)
    precision, recall = precision_recall(m)
    per_ill, ci_ill = {}, {}
    for name in ILLUMINATION:
        ids = (illumination or {}).get(name) or []
        if ids:
            per_ill[name] = evaluate_coco(gt, dets, ids)["map50"]
            ci_ill[name] = list(bootstrap_ci(subset(m, ids), n_boot, seed))
        else:
            per_ill[name], ci_ill[name] = None, [None, None]
    return {
        "map50": base["map50"],
        "map50_95": base["map50_95"],
        "precision": precision,
        "recall": recall,
        "per_class": base["per_class"],
        "per_illumination": per_ill,
        "per_size": base["per_size"],
        "ci95": {"map50": list(bootstrap_ci(m, n_boot, seed)), "per_illumination": ci_ill},
        "n_images": base["n_images"],
        "n_dets": base["n_dets"],
    }
```

"""baykus-eval: Ultralytics val → pycocotools metrikleri + güven aralığı → run JSON (FR-V)."""

from __future__ import annotations

import argparse
import contextlib
import io
import json
import platform
import sys
from pathlib import Path

from pycocotools.coco import COCO

from baykus.eval.metrics import build_metrics
from baykus.eval.predictions import load_predictions, remap_to_gt, save_predictions
from baykus.runinfo import (
    git_info,
    iso,
    make_run_id,
    model_family,
    model_size,
    software_versions,
    utc_now,
    write_json,
)


def _load_coco(path: Path) -> COCO:
    with contextlib.redirect_stdout(io.StringIO()):
        return COCO(str(path))


def _hardware() -> dict:
    try:
        import torch

        if torch.cuda.is_available():
            return {"device": torch.cuda.get_device_name(0), "driver": torch.version.cuda}
    except ImportError:
        pass
    return {"device": platform.processor() or platform.machine() or "cpu", "driver": None}


def _slices(meta: Path, split: str) -> dict | None:
    p = meta / f"slices_{split}.json"
    return json.loads(p.read_text(encoding="utf-8")).get("illumination") if p.exists() else None


def _cmd_run(a: argparse.Namespace) -> int:
    from ultralytics import YOLO

    from baykus.data.convert import activate_label_set
    from baykus.eval.coco_eval import MAX_DETS
    from baykus.train.cli import RECORD, resolve_data
    from baykus.train.config import load_config

    cfg = load_config(Path(a.config))
    weights = Path(a.weights)
    data = resolve_data(cfg, a.data_root)
    root = data.parent
    meta = Path(a.meta) if a.meta else root / "meta"
    rec_path = weights.parent.parent / RECORD
    train_rec = json.loads(rec_path.read_text(encoding="utf-8")) if rec_path.exists() else {}
    git = git_info()
    run_id = train_rec.get("run_id") or make_run_id(
        cfg.experiment, cfg.model, git["git_commit"], utc_now()
    )

    activate_label_set(root, cfg.modality, "ir")  # çapraz kontrol IR etiketleriyle (FR-V6)
    name = f"{run_id}_{a.split}"
    kwargs = dict(data=str(data), split=a.split, conf=0.001, max_det=MAX_DETS, save_json=True,
                  imgsz=cfg.train.imgsz, batch=cfg.train.batch, plots=False,
                  project=str(a.project), name=name, exist_ok=True)
    if a.device:
        kwargs["device"] = a.device
    ul = YOLO(str(weights)).val(**kwargs)
    save_dir = Path(a.project) / name
    pred_file = save_dir / "predictions.json"  # Ultralytics tahmin yoksa dosyayı hiç yazmaz
    ul_preds = json.loads(pred_file.read_text(encoding="utf-8")) if pred_file.exists() else []
    n_boot = a.n_boot if a.n_boot is not None else cfg.eval.n_boot

    for i, ls in enumerate(cfg.eval.label_sets):
        gt = _load_coco(root / "coco" / f"{a.split}_{ls}.json")
        dets, dropped = remap_to_gt(ul_preds, gt)
        if i == 0:
            save_predictions(save_dir / "predictions_coco.json.gz", dets)
        metrics = build_metrics(gt, dets, _slices(meta, a.split), n_boot=n_boot,
                                seed=cfg.train.seed)
        record = {
            "run_id": run_id, "git_commit": git["git_commit"], "git_dirty": git["git_dirty"],
            "config": a.config, "config_hash": cfg.config_hash,
            "mlflow_run_id": train_rec.get("mlflow_run_id"), "train_run_id": train_rec.get("run_id"),
            "seed": cfg.train.seed, "dataset": "dronevehicle", "split": a.split, "label_set": ls,
            "model": {"family": model_family(cfg.model), "size": model_size(cfg.model),
                      "mode": cfg.mode, "params_m": None, "gflops": None},
            "format": "pytorch", "precision": "fp32", "hardware": _hardware(),
            "software": software_versions(), "weights": str(weights), "n_boot": n_boot, "dropped_predictions": dropped,
            "metrics": metrics, "created_at": iso(utc_now()),
        }
        if ls == "ir":
            ul_map50 = float(ul.box.map50)
            record["crosscheck"] = {
                "label_set": "ir", "ultralytics_map50": ul_map50,
                "pycocotools_map50": metrics["map50"],
                "abs_diff": abs(ul_map50 - metrics["map50"]),
            }
        out = write_json(Path(a.results) / "runs" / f"{name}_{ls}.json", record)
        print(out, f"map50={metrics['map50']:.4f}", f"dropped={dropped}")
        if dropped:
            print(f"uyarı: {dropped} tahmin eşlenemedi", file=sys.stderr)
    return 0


def _cmd_compare(a: argparse.Namespace) -> int:
    from baykus.eval.bootstrap import collect_matches, paired_diff_ci, subset

    gt = _load_coco(Path(a.gt))
    ma = collect_matches(gt, load_predictions(Path(a.a)))
    mb = collect_matches(gt, load_predictions(Path(a.b)))
    out = {"name": a.name, "gt": a.gt, "a": a.a, "b": a.b, "n_boot": a.n_boot,
           "overall": paired_diff_ci(ma, mb, a.n_boot), "per_illumination": {}}
    slices = json.loads(Path(a.slices).read_text(encoding="utf-8")).get("illumination", {})
    for name, ids in slices.items():
        out["per_illumination"][name] = (
            paired_diff_ci(subset(ma, ids), subset(mb, ids), a.n_boot) if ids else None
        )
    out["created_at"] = iso(utc_now())
    print(write_json(Path(a.results) / "compare" / f"{a.name}.json", out))
    return 0


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="baykus-eval")
    sub = p.add_subparsers(dest="cmd", required=True)
    r = sub.add_parser("run", help="bir modeli değerlendir")
    r.add_argument("--config", required=True)
    r.add_argument("--weights", required=True)
    r.add_argument("--split", default="val", choices=["train", "val", "test"])
    r.add_argument("--data-root", type=Path, default=None)
    r.add_argument("--meta", type=Path, default=None)
    r.add_argument("--project", type=Path, default=Path("runs/eval"))
    r.add_argument("--results", type=Path, default=Path("../results"))
    r.add_argument("--n-boot", type=int, default=None)
    r.add_argument("--device", default=None)
    r.set_defaults(func=_cmd_run)
    c = sub.add_parser("compare", help="iki modelin eşleştirilmiş bootstrap farkı")

"""results/runs/*.json → markdown tablo. Eksik değer '—'; tahmini değer yazılmaz."""

from __future__ import annotations

import json
from pathlib import Path

ILLUMINATION = ("day", "night", "dark_night")
HEADER = (
    "| Çalıştırma | Mod | Split | Etiket | mAP50 [%95 GA] | mAP50-95 | Gündüz | Gece "
    "| Karanlık gece |\n|---|---|---|---|---|---|---|---|---|"
)


def load_runs(folder: Path, split: str | None = None, label_set: str | None = None) -> list[dict]:
    runs = [json.loads(p.read_text(encoding="utf-8")) for p in sorted(Path(folder).glob("*.json"))]
    runs = [r for r in runs if (split is None or r.get("split") == split)
            and (label_set is None or r.get("label_set") == label_set)]
    return sorted(runs, key=lambda r: r["run_id"])


def fmt(v: float | None, ci: list | None = None) -> str:
    if v is None:
        return "—"
    s = f"{100 * v:.1f}"
    if ci and None not in ci:
        s += f" [{100 * ci[0]:.1f}, {100 * ci[1]:.1f}]"
    return s


def main_table(runs: list[dict]) -> str:
    rows = [HEADER]
    for r in runs:
        m = r["metrics"]
        ill = m.get("per_illumination") or {}
        cells = [
            r["run_id"], r["model"]["mode"], r["split"], r["label_set"],
            fmt(m.get("map50"), (m.get("ci95") or {}).get("map50")),
            fmt(m.get("map50_95")),
            *(fmt(ill.get(k)) for k in ILLUMINATION),
        ]
        rows.append("| " + " | ".join(cells) + " |")
    return "\n".join(rows)
```

"""Aydınlatma kırılımı çubuk grafiği (güven aralığı hata çubuğu ile)."""

from __future__ import annotations

from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt  # noqa: E402
import numpy as np  # noqa: E402

from baykus.report.tables import ILLUMINATION  # noqa: E402

LABELS = {"day": "Gündüz", "night": "Gece", "dark_night": "Karanlık gece"}


def illumination_plot(runs: list[dict], path: Path) -> Path:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    fig, ax = plt.subplots(figsize=(7, 3.5))
    width = 0.8 / max(len(runs), 1)
    x = np.arange(len(ILLUMINATION))
    for i, r in enumerate(runs):
        m = r["metrics"]
        ill = m.get("per_illumination") or {}
        ci = (m.get("ci95") or {}).get("per_illumination") or {}
        vals, err = [], [[], []]
        for k in ILLUMINATION:
            v = ill.get(k)
            lo, hi = (ci.get(k) or [None, None])
            vals.append(np.nan if v is None else 100 * v)
            err[0].append(0 if v is None or lo is None else 100 * (v - lo))
            err[1].append(0 if v is None or hi is None else 100 * (hi - v))
        ax.bar(x + i * width, vals, width, yerr=err, capsize=3,
               label=f"{r['model']['mode']} ({r['label_set']})")
    ax.set_xticks(x + width * (len(runs) - 1) / 2, [LABELS[k] for k in ILLUMINATION])
    ax.set_ylabel("mAP50 (%)")
    ax.set_ylim(0, 100)
    ax.legend(fontsize=8)
    fig.tight_layout()
    fig.savefig(path, dpi=150)
    plt.close(fig)
    return path
