## Spec yorumları (belgelenir)

1. **Karartma (§12.3 "gamma ×0.5"):** piksel değeri çarpanla ölçeklenir, `out = round(img × f)`. Gamma eğrisi (`img^γ`) değil; "×" gösterimi ve "kamera daha az ışık alıyor" senaryosu çarpanla birebir örtüşüyor. `docs/robustness.md` başlığında yazar.
2. **Hizasızlık:** IR görüntüsü yatayda N piksel sağa kaydırılır, boşalan şerit 0, etiketler değişmez (etiketler nesnenin gerçek yeri). Sadece iki modaliteyi birden gören modellere uygulanır: tek modaliteli modelde kaydırma "hizasızlık" değil "etiket kayması" olur.
3. **Uygulanabilirlik:** Bir bozulma, modelin görmediği modaliteye aitse o model için koşturulmaz ve raporda "uygulanmaz" yazar (ör. E1 + IR gürültüsü). Sonuç zaten temiz skorla aynı olurdu.
4. **Bozulmuş RGB PNG olarak yazılır:** JPEG yeniden sıkıştırması kendi bozulmasını eklerdi. Temiz koşul aynı yoldan geçer ve ana val sonucuyla aynı çıkmalıdır (Görev 11 kontrolü).
5. **Sağlamlık paketinde iki etiket seti yok:** sadece IR (ana set). E5 sorusu temiz veride cevaplanıyor.

def _make_pairs(root: Path, split: str, stems: list[str]) -> None:
    rng = np.random.default_rng(len(split))

        rgb = rng.integers(0, 255, (16, 20, 3), dtype=np.uint8)

    with tifffile.TiffFile(p) as t:
        return np.stack([pg.asarray() for pg in t.pages], axis=-1)  # (H, W, 4)

    assert build_rgbt4(tmp_path, splits=["val"]) == 1
    got = _pages(tmp_path / "rgbt4/images/val/00001.tif")
    bgr = imread_any(tmp_path / "rgb/images/val/00001.jpg", cv2.IMREAD_COLOR)
    ir = imread_any(tmp_path / "ir/images/val/00001.png", cv2.IMREAD_GRAYSCALE)

`python/src/baykus/data/multichannel.py` (tamamı):
```python
"""RGB + termal → 4 kanallı TIFF (FR-D5). Biçim ADR 0004'te."""

from __future__ import annotations

from collections.abc import Iterable
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import cv2
import numpy as np
import tifffile

from baykus.data.layout import SPLITS
from baykus.io import imread_any

TIFF_LAYOUT = "pages"  # ADR 0004: her kanal ayrı gri sayfa (Spike A)


def stack_rgbt(bgr: np.ndarray, ir: np.ndarray) -> np.ndarray:
    if bgr.shape[:2] != ir.shape[:2]:
        raise ValueError(f"boyut uyuşmazlığı: {bgr.shape[:2]} vs {ir.shape[:2]}")
    ir2 = ir if ir.ndim == 2 else ir[..., 0]
    return np.dstack([bgr[..., 2], bgr[..., 1], bgr[..., 0], ir2]).astype(np.uint8)


def write_rgbt4(path: Path, rgbt: np.ndarray) -> None:
    """Her kanal ayrı gri sayfa. Tek sayfalı RGB+alfa TIFF'te OpenCV, RGB'yi 4. kanalla
    çarpıp BGR'ye çeviriyor (Spike A); çok sayfalıyı Ultralytics sırayla yığıyor."""
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with tifffile.TiffWriter(path) as tw:
        for c in range(rgbt.shape[2]):
            tw.write(rgbt[..., c], photometric="minisblack", compression="deflate")


def read_pair(rgb_path: Path, ir_path: Path) -> tuple[np.ndarray, np.ndarray]:
    """(BGR, IR gri). Eksik ya da okunamayan dosya adıyla OSError verir."""
    bgr = imread_any(rgb_path, cv2.IMREAD_COLOR)
    if bgr is None:
        raise OSError(f"okunamadı: {rgb_path}")
    ir = imread_any(ir_path, cv2.IMREAD_GRAYSCALE)
    if ir is None:
        raise OSError(f"okunamadı: {ir_path}")
    return bgr, ir


def _write_one(job: tuple[Path, Path, Path]) -> None:
    rgb_path, ir_path, out_path = job
    write_rgbt4(out_path, stack_rgbt(*read_pair(rgb_path, ir_path)))


def build_rgbt4(root: Path, splits: Iterable[str] = SPLITS, workers: int = 1) -> int:
    """rgb/ + ir/ → rgbt4/images/<split>/*.tif. ADR 0004: E3 oturumunda yeniden üretilir."""
    root = Path(root)
    jobs = [
        (p, root / "ir/images" / split / f"{p.stem}.png", root / "rgbt4/images" / split / f"{p.stem}.tif")
        for split in splits
        for p in sorted((root / "rgb/images" / split).glob("*.jpg"))
    ]
    if workers <= 1:
        for job in jobs:
            _write_one(job)
    else:
        with ProcessPoolExecutor(max_workers=workers) as ex:
            list(ex.map(_write_one, jobs, chunksize=32))
    return len(jobs)
```

    import time

"""Sağlamlık paketi bozulmaları (spec §12.3, FR-V4, kanca K2).

Saf numpy/OpenCV fonksiyonları: bölümden bağımsız ve seed'li, train'de de kullanılabilir.
RGB bellekte BGR uint8 (H, W, 3), IR gri uint8 (H, W). Gürültü görüntü başına seed'lenir
(seed + dosya adı + koşul): paralel işçi ya da sıra sonucu değiştirmez.
"""

from __future__ import annotations

import zlib
from collections.abc import Iterable
from dataclasses import dataclass

import cv2
import numpy as np

# ad → (hedef modalite, spec §12.3 şiddetleri; özdeş şiddet dahil)
SUITE: dict[str, tuple[str, tuple[float, ...]]] = {
    "darken": ("rgb", (1.0, 0.5, 0.25, 0.1)),  # çarpan (spec yorumu 1)
    "noise": ("ir", (0, 5, 10, 20)),  # σ, 8-bit
    "blur": ("both", (0, 3, 7, 11)),  # Gauss çekirdeği
    "misalign": ("ir", (0, 5, 10, 20, 40)),  # IR yatay kaydırma, piksel
    "drop_rgb": ("rgb", (1,)),  # kamera kaybı
    "drop_ir": ("ir", (1,)),
}
IDENTITY: dict[str, float] = {"darken": 1.0, "noise": 0, "blur": 0, "misalign": 0}
MODE_NEEDS: dict[str, set[str]] = {"rgb": {"rgb"}, "ir": {"ir"}, "early_fusion": {"rgb", "ir"}}


@dataclass(frozen=True)
class Condition:
    corruption: str  # clean | SUITE anahtarı
    severity: float
    target: str  # none | rgb | ir | both

    @property
    def id(self) -> str:
        if self.corruption in ("clean", "drop_rgb", "drop_ir"):
            return self.corruption
        return f"{self.corruption}-{self.severity:g}"


CLEAN = Condition("clean", 0.0, "none")


def darken(img: np.ndarray, factor: float) -> np.ndarray:
    if not 0.0 <= factor <= 1.0:
        raise ValueError(f"darken: çarpan 0-1 arası olmalı, {factor} verildi")
    return np.clip(np.rint(img.astype(np.float32) * factor), 0, 255).astype(np.uint8)


def gaussian_noise(img: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    if sigma < 0:
        raise ValueError(f"noise: σ negatif olamaz, {sigma} verildi")
    if sigma == 0:
        return img.copy()
    noisy = img.astype(np.float32) + rng.normal(0.0, sigma, img.shape).astype(np.float32)
    return np.clip(np.rint(noisy), 0, 255).astype(np.uint8)


def gaussian_blur(img: np.ndarray, ksize: int) -> np.ndarray:
    if ksize == 0:
        return img.copy()
    if ksize < 0 or ksize % 2 == 0:
        raise ValueError(f"blur: çekirdek pozitif tek sayı olmalı, {ksize} verildi")
    return cv2.GaussianBlur(img, (ksize, ksize), 0)


def shift(img: np.ndarray, dx: int, dy: int = 0) -> np.ndarray:
    """İçeriği (dx, dy) piksel kaydırır (pozitif dx sağa, dy aşağı); boşalan kenar 0."""
    h, w = img.shape[:2]
    out = np.zeros_like(img)
    if abs(dx) >= w or abs(dy) >= h:
        return out
    out[max(dy, 0) : h + min(dy, 0), max(dx, 0) : w + min(dx, 0)] = img[
        max(-dy, 0) : h - max(dy, 0), max(-dx, 0) : w - max(dx, 0)
    ]
    return out


def suite_conditions(only: Iterable[str] | None = None) -> list[Condition]:
    names = list(SUITE) if only is None else list(only)
    unknown = [n for n in names if n not in SUITE]
    if unknown:
        raise ValueError(f"bilinmeyen bozulma: {unknown}; seçenekler: {list(SUITE)}")
    conds = [CLEAN]
    for name in SUITE:  # sıra her zaman SUITE sırası
        if name not in names:
            continue
        target, sevs = SUITE[name]
        conds += [Condition(name, float(s), target) for s in sevs if s != IDENTITY.get(name)]
    return conds


def applies(cond: Condition, mode: str) -> bool:
    needs = MODE_NEEDS[mode]
    if cond.corruption == "clean":
        return True
    if cond.corruption == "misalign":  # tek modalitede kaydırma = etiket kayması (spec yorumu 2)
        return needs == {"rgb", "ir"}
    touched = {"rgb", "ir"} if cond.target == "both" else {cond.target}
    return bool(needs & touched)


def image_rng(seed: int, stem: str, cond: Condition) -> np.random.Generator:
    return np.random.default_rng([seed, zlib.crc32(f"{stem}|{cond.id}".encode())])


def apply_condition(
    bgr: np.ndarray, ir: np.ndarray, cond: Condition, *, seed: int, stem: str
) -> tuple[np.ndarray, np.ndarray]:
    c, s = cond.corruption, cond.severity
    if c == "clean":
        return bgr.copy(), ir.copy()
    if c == "darken":
        return darken(bgr, s), ir.copy()
    if c == "noise":
        return bgr.copy(), gaussian_noise(ir, s, image_rng(seed, stem, cond))
    if c == "blur":
        return gaussian_blur(bgr, int(s)), gaussian_blur(ir, int(s))
    if c == "misalign":
        return bgr.copy(), shift(ir, int(s), 0)
    if c == "drop_rgb":
        return np.zeros_like(bgr), ir.copy()
    if c == "drop_ir":
        return bgr.copy(), np.zeros_like(ir)
    raise ValueError(f"bilinmeyen bozulma: {c}")
```

"""Bir bozulma koşulu için değerlendirme klasörü (FR-V4, kancalar K1-K2).

Düzen işlenmiş verinin aynısı: <out>/{rgb,ir,rgbt4}/images/<split>, labels_<set>/<split>,
coco/<split>_<set>.json, data_*.yaml. Böylece baykus-eval aynı yoldan ölçer.
Bozulmuş RGB PNG yazılır: JPEG yeniden sıkıştırması kendi bozulmasını eklerdi (spec yorumu 4).
"""

from __future__ import annotations

import shutil
from collections.abc import Iterable
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

from baykus.data.multichannel import read_pair, stack_rgbt, write_rgbt4
from baykus.data.yamls import write_data_yamls
from baykus.io import imwrite_any
from baykus.robustness.corruptions import Condition, apply_condition

MODALITIES = ("rgb", "ir", "rgbt4")


def _write_one(job: tuple) -> None:
    rgb_path, ir_path, out, split, stem, cond, seed, mods = job
    bgr, ir = apply_condition(*read_pair(rgb_path, ir_path), cond, seed=seed, stem=stem)
    if "rgb" in mods:
        imwrite_any(out / "rgb/images" / split / f"{stem}.png", bgr)
    if "ir" in mods:
        imwrite_any(out / "ir/images" / split / f"{stem}.png", ir)
    if "rgbt4" in mods:
        write_rgbt4(out / "rgbt4/images" / split / f"{stem}.tif", stack_rgbt(bgr, ir))


def make_condition(
    src: Path,
    out: Path,
    split: str,
    cond: Condition,
    modalities: Iterable[str],
    seed: int,
    workers: int = 1,
    label_set: str = "ir",
) -> int:
    src, out = Path(src), Path(out)
    mods = tuple(sorted(set(modalities)))
    unknown = [m for m in mods if m not in MODALITIES]
    if unknown:
        raise ValueError(f"bilinmeyen modalite: {unknown}; seçenekler: {MODALITIES}")
    stems = sorted(p.stem for p in (src / "rgb/images" / split).glob("*.jpg"))
    jobs = [
        (
            src / "rgb/images" / split / f"{s}.jpg",
            src / "ir/images" / split / f"{s}.png",
            out,
            split,
            s,
            cond,
            seed,
            mods,
        )
        for s in stems
    ]
    if workers <= 1:
        for job in jobs:
            _write_one(job)
    else:
        with ProcessPoolExecutor(max_workers=workers) as ex:
            list(ex.map(_write_one, jobs, chunksize=32))
    labels = src / f"labels_{label_set}" / split
    if labels.is_dir():
        shutil.copytree(labels, out / f"labels_{label_set}" / split, dirs_exist_ok=True)
    coco = src / "coco" / f"{split}_{label_set}.json"
    if coco.exists():
        (out / "coco").mkdir(parents=True, exist_ok=True)
        shutil.copy2(coco, out / "coco" / coco.name)

from __future__ import annotations

import json
import sys
import types
from pathlib import Path
from types import SimpleNamespace

import yaml


class FakeYOLO:
    calls: list[dict] = []

    def __init__(self, weights: str):
        self.weights = weights

    def val(self, **kw):
        FakeYOLO.calls.append(kw)
        data = Path(kw["data"])
        cfg = yaml.safe_load(data.read_text(encoding="utf-8"))
        img_dir = Path(cfg["path"]) / cfg[kw["split"]]
        if not any(img_dir.glob("*")):
            raise FileNotFoundError(f"görüntü yok: {img_dir}")

            {
                "image_id": int(stem[a["image_id"]]),
                "category_id": a["category_id"],
                "bbox": a["bbox"],
                "score": 0.9,
            }
            for a in gt["annotations"]
        ]
        out = Path(kw["project"]) / kw["name"]
        out.mkdir(parents=True, exist_ok=True)
        (out / "predictions.json").write_text(json.dumps(preds), encoding="utf-8")
        return SimpleNamespace(box=SimpleNamespace(map50=1.0))


def install(monkeypatch) -> type:
    FakeYOLO.calls = []
    mod = types.ModuleType("ultralytics")

    return FakeYOLO

    (Path(run_dir) / "baykus_train.json").write_text(json.dumps(rec), encoding="utf-8")

def train_record(weights: Path) -> dict:
    """<run>/weights/best.pt → <run>/baykus_train.json (yoksa {})."""
    from baykus.train.cli import RECORD

    p = Path(weights).parent.parent / RECORD
    return json.loads(p.read_text(encoding="utf-8")) if p.exists() else {}


def evaluate_weights(
    cfg: ExperimentConfig,
    *,
    config: str,
    weights: Path,
    data: Path,
    split: str,
    meta: Path,
    project: Path,
    results: Path,

    label_sets: tuple[str, ...] | None = None,

    """Ultralytics val → etiket seti başına run JSON. condition verilirse (sağlamlık paketi)
    ada koşul kimliği eklenir ve kayda `robustness` alanı yazılır."""
    from ultralytics import YOLO

    from baykus.data.convert import activate_label_set
    from baykus.eval.coco_eval import MAX_DETS

    weights, data, meta = Path(weights), Path(data), Path(meta)
    root = data.parent
    train_rec = train_record(weights)
    git = git_info()
    run_id = train_rec.get("run_id") or make_run_id(
        cfg.experiment, cfg.model, git["git_commit"], utc_now()
    )

    activate_label_set(root, cfg.modality, "ir")  # çapraz kontrol IR etiketleriyle (FR-V6)
    name = f"{run_id}_{split}" + (f"_{condition['id']}" if condition else "")

        save_json=True,

        batch=cfg.train.batch,
        plots=False,
        project=str(project),
        name=name,
        exist_ok=True,
    )

        kwargs["device"] = device

            "git_commit": git["git_commit"],
            "git_dirty": git["git_dirty"],

            "config_hash": cfg.config_hash,
            "mlflow_run_id": train_rec.get("mlflow_run_id"),
            "train_run_id": train_rec.get("run_id"),
            "seed": cfg.train.seed,
            "dataset": "dronevehicle",
            "split": split,
            "label_set": ls,
            "model": {
                "family": model_family(cfg.model),
                "size": model_size(cfg.model),
                "mode": cfg.mode,
                "params_m": None,
                "gflops": None,
            },
            "format": "pytorch",
            "precision": "fp32",
            "hardware": hardware_info(),
            "software": software_versions(),
            "weights": str(weights),
            "n_boot": n_boot,
            "dropped_predictions": dropped,
            "metrics": metrics,
            "created_at": iso(utc_now()),
        }
        if condition is not None:
            record["robustness"] = condition
        if ls == "ir":
            ul_map50 = float(ul.box.map50)
            record["crosscheck"] = {
                "label_set": "ir",
                "ultralytics_map50": ul_map50,
                "pycocotools_map50": metrics["map50"],
                "abs_diff": abs(ul_map50 - metrics["map50"]),
            }
        out = write_json(Path(results) / results_subdir / f"{name}_{ls}.json", record)
        print(out, f"map50={metrics['map50']:.4f}", f"dropped={dropped}")
        if dropped:
            print(f"uyarı: {dropped} tahmin eşlenemedi", file=sys.stderr)
        outs.append(out)
    return outs

        results=a.results,
        n_boot=a.n_boot if a.n_boot is not None else cfg.eval.n_boot,
        device=a.device,
    )
    return 0

"""baykus-robust: sağlamlık paketi (spec §12.3, FR-V4, kanca K1).

Koşul koşul ilerler: bozulmuş klasörü üret → koşulun uygulandığı her modeli değerlendir →
klasörü sil. Sonuç JSON'u olan (model, koşul) atlanır: oturum koparsa aynı komut kaldığı
yerden sürer. Bütün modeller aynı seed'le bozulmuş aynı pikselleri görür.
"""

from __future__ import annotations

import argparse
import shutil
import sys
from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path

from baykus.robustness.corruptions import SUITE, Condition, applies, suite_conditions
from baykus.robustness.dataset import make_condition
from baykus.train.config import MODALITY_DIR, ExperimentConfig, load_config

ROBUST_DIR = "robust"
LABEL_SET = "ir"  # ana etiket seti (ADR 0003); sağlamlık sadece onunla ölçülür


class RobustError(ValueError):
    pass


@dataclass(frozen=True)
class ModelSpec:
    config: str
    weights: Path
    cfg: ExperimentConfig
    run_id: str


def load_model_spec(arg: str) -> ModelSpec:
    from baykus.eval.cli import train_record

    if "=" not in arg:
        raise RobustError(f"--model CONFIG=AĞIRLIK biçiminde olmalı (= yok): {arg!r}")
    config, w = arg.split("=", 1)
    weights = Path(w)
    run_id = train_record(weights).get("run_id")
    if not run_id:
        raise RobustError(f"{weights}: eğitim kaydı (baykus_train.json) yok; run_id bilinmiyor")
    return ModelSpec(config, weights, load_config(Path(config)), run_id)


def result_path(results: Path, m: ModelSpec, split: str, cond: Condition) -> Path:
    return Path(results) / ROBUST_DIR / f"{m.run_id}_{split}_{cond.id}_{LABEL_SET}.json"


def plan(
    models: list[ModelSpec], conds: list[Condition], results: Path, split: str
) -> list[tuple[Condition, list[ModelSpec]]]:
    out = []
    for c in conds:
        todo = [
            m
            for m in models
            if applies(c, m.cfg.mode) and not result_path(results, m, split, c).exists()
        ]
        if todo:
            out.append((c, todo))
    return out


def run_suite(
    models: list[ModelSpec],
    conds: list[Condition],
    *,
    src: Path,
    work: Path,
    meta: Path,
    split: str,
    project: Path,
    results: Path,
    seed: int,
    n_boot: int,
    device: str | None = None,
    workers: int = 1,
    evaluate: Callable | None = None,
) -> int:
    if evaluate is None:
        from baykus.eval.cli import evaluate_weights

        evaluate = evaluate_weights
    n = 0
    for cond, todo in plan(models, conds, results, split):
        cdir = Path(work) / cond.id
        shutil.rmtree(cdir, ignore_errors=True)  # önceki oturumdan yarım kalmış olabilir
        mods = {MODALITY_DIR[m.cfg.mode] for m in todo}
        make_condition(src, cdir, split, cond, mods, seed, workers)
        info = {
            "id": cond.id,
            "corruption": cond.corruption,
            "severity": cond.severity,
            "target": cond.target,
            "seed": seed,
        }
        try:
            for m in todo:
                evaluate(
                    m.cfg,
                    config=m.config,
                    weights=m.weights,
                    data=cdir / Path(m.cfg.data).name,
                    split=split,
                    meta=meta,
                    project=project,
                    results=results,
                    n_boot=n_boot,
                    device=device,
                    label_sets=(LABEL_SET,),
                    condition=info,
                    results_subdir=ROBUST_DIR,
                )
                # Ultralytics'in ham tahmin kopyası Drive'da yer kaplar; bizimki .json.gz'de (K1)
                (Path(project) / f"{m.run_id}_{split}_{cond.id}" / "predictions.json").unlink(
                    missing_ok=True
                )
                n += 1
                print(f"[{cond.id}] {m.cfg.experiment} tamam", flush=True)
        finally:
            shutil.rmtree(cdir, ignore_errors=True)
    return n


def _cmd_run(a: argparse.Namespace) -> int:
    try:
        models = [load_model_spec(s) for s in a.model]
    except (RobustError, FileNotFoundError) as e:
        print(f"hata: {e}", file=sys.stderr)
        return 2
    n = run_suite(
        models,
        suite_conditions(a.only),
        src=Path(a.data_root),
        work=Path(a.work),
        meta=Path(a.meta),
        split=a.split,
        project=Path(a.project),
        results=Path(a.results),
        seed=a.seed,
        n_boot=a.n_boot if a.n_boot is not None else models[0].cfg.eval.n_boot,
        device=a.device,
        workers=a.workers,
    )
    print(f"değerlendirilen (model, koşul): {n}")
    return 0


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="baykus-robust")
    sub = p.add_subparsers(dest="cmd", required=True)
    r = sub.add_parser("run", help="sağlamlık paketini çalıştır (kaldığı yerden sürer)")
    r.add_argument("--model", action="append", required=True, help="CONFIG=AĞIRLIK, tekrarlanır")
    r.add_argument("--data-root", required=True, help="işlenmiş veri (rgb/, ir/, labels_ir/, coco/)")
    r.add_argument("--meta", required=True, help="slices_<split>.json klasörü")
    r.add_argument("--split", default="val", choices=["val", "test"])
    r.add_argument("--work", default="runs/robust_work", help="geçici koşul klasörleri")
    r.add_argument("--project", default="runs/robust", help="tahmin dosyaları (K1)")
    r.add_argument("--results", default="../results")

    r.add_argument("--n-boot", type=int, default=None)
    r.add_argument("--device", default=None)
    r.add_argument("--workers", type=int, default=1)
    r.set_defaults(func=_cmd_run)
    return p


def main(argv: list[str] | None = None) -> int:
    a = build_parser().parse_args(argv)
    return a.func(a)

    "| Çalıştırma | Mod | mAP50 IR [%95 GA] | mAP50 RGB [%95 GA] | Δ (RGB − IR) | Δ gündüz "

def _delta(a: float | None, b: float | None) -> str:
    return "—" if a is None or b is None else f"{100 * (b - a):+.1f}"


def label_set_table(runs: list[dict]) -> str:
    """E5: aynı tahminler iki etiket setinde (FR-V3). Δ = RGB − IR, puan."""
    pairs: dict[tuple[str, str], dict[str, dict]] = {}
    for r in runs:
        pairs.setdefault((r["run_id"], r["split"]), {})[r["label_set"]] = r
    rows = [LABEL_SET_HEADER]
    for (run_id, _), p in sorted(pairs.items()):
        if "ir" not in p or "rgb" not in p:
            continue
        a, b = p["ir"]["metrics"], p["rgb"]["metrics"]
        ia, ib = a.get("per_illumination") or {}, b.get("per_illumination") or {}
        cells = [
            run_id,
            p["ir"]["model"]["mode"],
            fmt(a.get("map50"), (a.get("ci95") or {}).get("map50")),
            fmt(b.get("map50"), (b.get("ci95") or {}).get("map50")),
            _delta(a.get("map50"), b.get("map50")),
            *(_delta(ia.get(k), ib.get(k)) for k in ILLUMINATION),
        ]
        rows.append("| " + " | ".join(cells) + " |")
    return "\n".join(rows)


def _diff(d: dict | None) -> str:
    if not d or d.get("diff") is None:
        return "—"
    lo, hi = d["ci95"]
    s = f"{100 * d['diff']:+.1f} [{100 * lo:+.1f}, {100 * hi:+.1f}]"
    return s + (" *" if d.get("significant") else "")


def compare_table(compares: list[dict]) -> str:
    """FR-V8 eşleştirilmiş bootstrap; fark = A − B (puan), * = aralık 0'ı içermiyor."""
    rows = [COMPARE_HEADER]
    for c in sorted(compares, key=lambda c: c["name"]):
        ill = c.get("per_illumination") or {}
        cells = [c["name"], _diff(c.get("overall")), *(_diff(ill.get(k)) for k in ILLUMINATION)]
        rows.append("| " + " | ".join(cells) + " |")
    return "\n".join(rows)


def load_compares(folder: Path) -> list[dict]:
    folder = Path(folder)
    if not folder.is_dir():
        return []
    return [json.loads(p.read_text(encoding="utf-8")) for p in sorted(folder.glob("*.json"))]

"—" ölçülmedi, "uygulanmaz" bozulma modelin görmediği modaliteye ait (spec yorumu 3).
"""

from __future__ import annotations

import json
from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt  # noqa: E402

from baykus.report.tables import fmt  # noqa: E402
from baykus.robustness.corruptions import IDENTITY, SUITE, Condition, applies  # noqa: E402

CURVES = ("darken", "noise", "blur", "misalign")
TITLES = {
    "darken": "Karartma (RGB, piksel çarpanı)",
    "noise": "Gauss gürültüsü (IR, σ)",
    "blur": "Bulanıklık (RGB + IR, Gauss çekirdeği)",
    "misalign": "Hizasızlık (IR yatay kaydırma, piksel)",
}
NA = "uygulanmaz"


def load_robust(folder: Path) -> list[dict]:
    recs = [json.loads(p.read_text(encoding="utf-8")) for p in sorted(Path(folder).glob("*.json"))]
    return [r for r in recs if "robustness" in r]


def model_key(r: dict) -> str:
    return Path(r["config"]).stem


def _index(recs: list[dict]) -> dict[tuple[str, str], dict]:
    return {(model_key(r), r["robustness"]["id"]): r for r in recs}


def _cell(r: dict | None) -> str:
    if r is None:
        return "—"
    m = r["metrics"]
    return fmt(m.get("map50"), (m.get("ci95") or {}).get("map50"))


def _cond_id(corruption: str, sev: float) -> str:
    if sev == IDENTITY.get(corruption):
        return "clean"
    return Condition(corruption, float(sev), SUITE[corruption][0]).id


def _applies_any(corruption: str, mode: str) -> bool:
    target, sevs = SUITE[corruption]
    return applies(Condition(corruption, float(sevs[-1]), target), mode)


def robust_markdown(recs: list[dict], fig_rel: str) -> str:
    idx = _index(recs)
    models = sorted({model_key(r) for r in recs})
    modes = {model_key(r): r["model"]["mode"] for r in recs}
    head = "| Şiddet | " + " | ".join(models) + " |\n|---|" + "---|" * len(models)
    out = [
        f"# Sağlamlık paketi ({recs[0]['split']}, IR etiketleri)",
        "",
        "> Bu dosya `baykus-robust report` ile `results/robust/*.json`'dan üretildi; elle "
        "düzenlenmez.",
        "> Hücre: mAP50 [görüntü düzeyinde bootstrap %95 GA]. `—` ölçülmedi; `uygulanmaz` "
        "bozulma modelin görmediği modaliteye ait ya da tek modaliteli modelde hizasızlık.",
        "> Karartma piksel değerini çarpanla ölçekler (gamma eğrisi değil). Hizasızlıkta IR "
        "kaydırılır, etiketler yerinde kalır.",
        "",
    ]
    for c in CURVES:
        out += [f"## {TITLES[c]}", "", head]
        for s in SUITE[c][1]:
            label = f"{s:g} (temiz)" if s == IDENTITY[c] else f"{s:g}"
            cells = [
                _cell(idx.get((mk, _cond_id(c, s)))) if _applies_any(c, modes[mk]) else NA
                for mk in models
            ]
            out.append(f"| {label} | " + " | ".join(cells) + " |")
        out.append("")
    out += ["## Kamera kaybı", "", head.replace("Şiddet", "Koşul", 1)]
    for label, cid, corr in (("Temiz", "clean", None), ("RGB kaybı", "drop_rgb", "drop_rgb"),
                             ("IR kaybı", "drop_ir", "drop_ir")):  # fmt: skip
        cells = [
            _cell(idx.get((mk, cid))) if corr is None or _applies_any(corr, modes[mk]) else NA
            for mk in models
        ]
        out.append(f"| {label} | " + " | ".join(cells) + " |")
    out += ["", f"![Sağlamlık eğrileri]({fig_rel})", ""]
    return "\n".join(out)


def robust_plot(recs: list[dict], path: Path) -> Path:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    idx = _index(recs)
    models = sorted({model_key(r) for r in recs})
    modes = {model_key(r): r["model"]["mode"] for r in recs}
    fig, axes = plt.subplots(2, 2, figsize=(9, 6.5))
    for ax, c in zip(axes.flat, CURVES, strict=True):
        for mk in models:
            if not _applies_any(c, modes[mk]):
                continue
            xs, ys, lo, hi = [], [], [], []
            for s in SUITE[c][1]:
                r = idx.get((mk, _cond_id(c, s)))
                if r is None or r["metrics"].get("map50") is None:
                    continue
                v = 100 * r["metrics"]["map50"]
                ci = (r["metrics"].get("ci95") or {}).get("map50") or [None, None]
                xs.append(s)
                ys.append(v)
                lo.append(v if ci[0] is None else 100 * ci[0])
                hi.append(v if ci[1] is None else 100 * ci[1])
            if xs:
                ax.plot(xs, ys, marker="o", label=mk)
                ax.fill_between(xs, lo, hi, alpha=0.15)
        ax.set_title(TITLES[c], fontsize=9)
        ax.set_ylim(0, 100)
        ax.set_ylabel("mAP50 (%)")
        if c == "darken":
            ax.invert_xaxis()  # sağa doğru şiddet artar
        if ax.get_legend_handles_labels()[0]:
            ax.legend(fontsize=7)
    fig.tight_layout()
    fig.savefig(path, dpi=150)
    plt.close(fig)
    return path
```

def _cmd_report(a: argparse.Namespace) -> int:
    from baykus.report.robust import load_robust, robust_markdown, robust_plot

    recs = load_robust(Path(a.robust))
    if not recs:
        print(f"hata: {a.robust} altında sağlamlık sonucu yok", file=sys.stderr)
        return 2
    out, fig = Path(a.out), Path(a.fig)
    robust_plot(recs, fig)
    rel = Path(os.path.relpath(fig, out.parent)).as_posix()
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(robust_markdown(recs, rel), encoding="utf-8")
    print(out)
    return 0
