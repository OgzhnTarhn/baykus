## Spec yorumları (belgelenir)

1. **WBF ızgarası (E4):** `w_rgb = 1` sabit, `ir_weight ∈ {0.5, 1, 2, 3}`, `iou_thr ∈ {0.5, 0.6, 0.7}`, `skip_box_thr = 0.001` (değerlendirmenin `conf=0.001`'iyle aynı: AP eğrisi kırpılmasın), `conf_type = "avg"`. WBF skoru ağırlıkların ölçeğinden bağımsız (`ortalama(s·w) · k / Σw`), sadece oran önemli; `w_rgb`'yi 1'e sabitlemek bir şey kaybettirmez.
2. **Seçim ölçütü:** val, IR etiketleri, mAP50 (ana metrik). Eşitlikte ızgara sırasındaki ilk değer. Seçilen parametrelerle ölçülen val sonucu **iyimser yanlıdır** (val'de seçildi, val'de raporlandı); deney günlüğünde yazılır, tarafsız sayı H5 test ölçümünde.
3. **E4b:** IoU eşiği E4'ün seçtiği; her aydınlatma diliminde (gündüz / gece / karanlık gece) `ir_weight` o dilimin mAP50'sini en yüksek yapan değer. Çıkarımda görüntünün aydınlatma etiketi RGB parlaklığından (FR-V7, FR-D4 kuralı, ADR 0005 eşikleri).
4. **Geç füzyonun sağlamlık ölçümü:** ayrı GPU koşusu yok; E1 ve E2'nin aynı koşuldaki tahminleri birleştirilir. Bozulma bir modaliteye dokunmuyorsa o modelin temiz tahmini kullanılır. Hizasızlık için E2 kaydırılmış IR'de ayrıca değerlendirilir (`baykus-robust run --ir-misalign`); bu E2 sonuçları tek başına "etiket kayması" olduğu için sağlamlık tablosunun E2 sütununda gösterilmez.
5. **E4b'nin bozulmuş görüntüdeki aydınlatma etiketi:** çıkarımda RGB'den türetildiği için bozulmadan etkilenir. Parlaklık (HSV V ortalaması) karartmada çarpanla ölçeklenir (yuvarlama ±0,5 gri seviye, testle sınırlı), RGB kaybında 0 olur (her kare karanlık gece → IR ağırlığı); bulanıklık, gürültü ve hizasızlıkta değişmez kabul edilir (bulanıklık ortalamayı kenar şeridi dışında korur; gürültü ve kaydırma IR'de).
6. **M3OT sınıfsız değerlendirme (§8.3):** Ultralytics val `single_cls=True` (tahmin ve etiket sınıfı 0) + `agnostic_nms=True`. İkincisi şart: YOLO26 uçtan uca başlığında top-k seçimi varsayılan olarak sınıf sınıf yapılır (`ultralytics/nn/modules/head.py`, `agnostic_nms`), sınıfsız değerlendirmede aynı araç için kalan ikinci sınıf kutusu yanlış pozitif sayılırdı. pycocotools tarafında bütün tahminler `category_id = 1`. Data yaml'ı 5 sınıf adını korur (model `nc=5`). İki etiket seti de ölçülür; tablo rgb modunu RGB etiketiyle, diğerlerini IR etiketiyle gösterir.
7. **M3OT bölümü ve kırılım:** modeller M3OT'ta hiçbir seçim yapmadığı için doğrudan **test** bölümü kullanılır. Aydınlatma kırılımı yok (M3OT'ta etiket yok, DroneVehicle eşikleri başka kamerada geçersiz) → `per_illumination` `null`. E4b M3OT'ta uygulanmaz; geç füzyon satırı E4 parametreleriyle.
8. **M3OT etiket süzgeci:** resmi `tools/convert_M3OT_to_yolo.py` ile aynı: `flag == 1`, `sınıf == 1`, `görünürlük > 0.25`; koordinat 0 tabanlı kabul edilir (resmi araç 1 çıkarmıyor), görüntüye kırpılır. KAPI M'de çizimle gözle doğrulanır.
9. **E3b `random` başlatma** yapılmaz (spec "bütçe varsa"). **E10** yapılmaz, gerekçesiyle not düşülür (Görev 17).

```python
class FakeYOLO:
    calls: list[dict] = []
    write_preds = True  # False: gerçek Ultralytics gibi tahmin yoksa dosya yazılmaz
    force_category: int | None = None  # sınıfsız testleri için: bütün tahminler bu sınıfta

    def __init__(self, weights: str):
        self.weights = weights

    def val(self, **kw):
        FakeYOLO.calls.append(kw)
        data = Path(kw["data"])
        cfg = yaml.safe_load(data.read_text(encoding="utf-8"))
        img_dir = Path(cfg["path"]) / cfg[kw["split"]]
        if not any(img_dir.glob("*")):
            raise FileNotFoundError(f"görüntü yok: {img_dir}")
        out = Path(kw["project"]) / kw["name"]
        out.mkdir(parents=True, exist_ok=True)
        if not FakeYOLO.write_preds:
            return SimpleNamespace(box=SimpleNamespace(map50=0.0))
        gt = json.loads(
            (data.parent / "coco" / f"{kw['split']}_ir.json").read_text(encoding="utf-8")
        )
        stem = {im["id"]: Path(im["file_name"]).stem for im in gt["images"]}
        preds = [
            {
                # Ultralytics: sayısal adda int, değilse ad (M3OT: seq_000001)
                "image_id": int(s) if (s := stem[a["image_id"]]).isdigit() else s,
                "category_id": FakeYOLO.force_category or a["category_id"],
                "bbox": a["bbox"],
                "score": 0.9,
            }
            for a in gt["annotations"]
        ]
        (out / "predictions.json").write_text(json.dumps(preds), encoding="utf-8")
        return SimpleNamespace(box=SimpleNamespace(map50=1.0))


def install(monkeypatch) -> type:
    FakeYOLO.calls = []
    FakeYOLO.write_preds = True
    FakeYOLO.force_category = None
    mod = types.ModuleType("ultralytics")
    mod.YOLO = FakeYOLO
    monkeypatch.setitem(sys.modules, "ultralytics", mod)
    return FakeYOLO
```

```python
def _mismatch(p: Path, seed: int | None, n_boot: int | None) -> str | None:
    rec = json.loads(p.read_text(encoding="utf-8"))
    got = (rec.get("robustness") or {}).get("seed")
    if seed is not None and got != seed:
        return f"seed {got} ≠ {seed}"
    if n_boot is not None and rec.get("n_boot") != n_boot:
        return f"n_boot {rec.get('n_boot')} ≠ {n_boot}"
    return None


def plan(
    models: list[ModelSpec],
    conds: list[Condition],
    results: Path,
    split: str,
    *,
    seed: int | None = None,
    n_boot: int | None = None,
) -> list[tuple[Condition, list[ModelSpec]]]:
    """Koşulun uygulandığı ve sonucu olmayan modeller. Var olan sonuç başka seed/n_boot ile
    yazılmışsa hata: iki ayar aynı tabloda sessizce karışmasın."""
    out = []
    for c in conds:
        todo = []
        for m in models:
            if not applies(c, m.cfg.mode):
                continue
            p = result_path(results, m, split, c)
            if p.exists():
                why = _mismatch(p, seed, n_boot)
                if why:
                    raise RobustError(
                        f"{p.name}: önceki sonuç başka ayarla ({why}); aynı ayarı kullan "
                        "ya da dosyayı sil"
                    )
                continue
            todo.append(m)
        if todo:
            out.append((c, todo))
    return out
```

ensemble-boxes `avg` kuralı (kaynak: `ensemble_boxes_wbf.py` 1.0.9, testlerin dayanağı):
- her kutunun ağırlıklı skoru `s·w`; kümenin kutusu `Σ(s·w·kutu) / Σ(s·w)`
- küme skoru `ortalama(s·w) · min(model sayısı, kümedeki kutu) / Σw`
- kutu bir kümeye `IoU > iou_thr` ise girer; farklı sınıflar birleşmez; koordinatlar 0-1 aralığında `x1, y1, x2, y2` beklenir.

```python
"""WBF sarmalayıcısı (FR-V7). (S) görevi: implementasyonu kullanıcı yazar, Claude test yazar.

Bir görüntünün model başına COCO tahminleri → ensemble-boxes weighted_boxes_fusion → COCO
tahminleri. Kütüphane 0-1 aralığında x1, y1, x2, y2 ister; dönüşüm ve doğrulama burada.
"""

from __future__ import annotations

from collections.abc import Sequence


def fuse_image(
    dets_per_model: Sequence[Sequence[dict]],
    weights: Sequence[float],
    *,
    width: int,
    height: int,
    iou_thr: float,
    skip_box_thr: float = 0.0,
) -> list[dict]:
    """Bir görüntünün tahminlerini birleştirir.

    dets_per_model[k]: k. modelin bu görüntüdeki tahminleri; her biri
        {"category_id": int, "bbox": [x, y, w, h] (piksel), "score": float}.
    weights[k]: k. modelin ağırlığı.
    Dönüş: {"category_id", "bbox" [x, y, w, h] piksel, "score"} listesi, skora göre azalan.

    Sözleşme (tests/test_wbf.py):
    - Ağırlık sayısı model sayısından farklıysa ya da bir ağırlık <= 0 ise ValueError
      (ensemble-boxes yanlış sayıda ağırlığı sessizce 1'e çeviriyor).
    - Kutu görüntüye kırpılır; kırpınca alanı 0 kalan kutu ve skoru <= 0 olan kutu atılır
      (kütüphane uyarı basmasın, 0'a bölüp NaN üretmesin).
    - conf_type="avg", allows_overflow=False (kütüphane varsayılanı).
    - Girdi listeleri değiştirilmez; çıktıda sayılar Python int/float.
    - Hiç kutu yoksa [].
    """
    raise NotImplementedError("(S) görevi: implementasyonu kullanıcı yazar")
```

```python
"""WBF sarmalayıcısı (S): elle hesaplanmış değerler (kural: Görev 4 başlığı, ensemble-boxes 1.0.9)."""

import warnings

import pytest

pytest.importorskip("ensemble_boxes")

from baykus.fusion.wbf import fuse_image  # noqa: E402

W, H = 100, 50


def det(x, y, w, h, s, c=1):
    return {"category_id": c, "bbox": [x, y, w, h], "score": s}


def fuse(a, b, weights=(1, 1), iou_thr=0.55, skip=0.0):
    return fuse_image(
        [a, b], list(weights), width=W, height=H, iou_thr=iou_thr, skip_box_thr=skip
    )


def test_identical_boxes_equal_weights_average_score():
    out = fuse([det(10, 10, 20, 10, 0.8)], [det(10, 10, 20, 10, 0.6)])
    assert len(out) == 1
    assert out[0]["score"] == pytest.approx(0.7, abs=1e-5)  # (0.8 + 0.6) / 2 · 2/2
    assert out[0]["bbox"] == pytest.approx([10, 10, 20, 10], abs=1e-3)
    assert out[0]["category_id"] == 1


def test_box_from_one_model_is_scaled_by_its_weight_share():
    assert fuse([det(10, 10, 20, 10, 0.8)], [])[0]["score"] == pytest.approx(0.4, abs=1e-5)
    # K4: ağırlık oranı skoru belirler: 0.8 · 2/3 ve 0.6 · 1/3
    one = fuse([det(10, 10, 20, 10, 0.8)], [], (2, 1))
    other = fuse([], [det(10, 10, 20, 10, 0.6)], (2, 1))
    assert one[0]["score"] == pytest.approx(0.5333, abs=1e-4)
    assert other[0]["score"] == pytest.approx(0.2, abs=1e-5)


def test_weighted_coordinates_and_score():
    # A = [0,0,10,10], B = [1,1,11,11] (x1y1x2y2); IoU 81/119 = 0.68 > 0.55 → tek küme.
    # Ağırlıklı skorlar 1.6 ve 0.6 → x1 = 0.6·1 / 2.2 = 0.2727; skor = 1.1 · 2/3 = 0.7333
    out = fuse([det(0, 0, 10, 10, 0.8)], [det(1, 1, 10, 10, 0.6)], (2, 1))
    assert len(out) == 1
    assert out[0]["bbox"] == pytest.approx([0.2727, 0.2727, 10.0, 10.0], abs=1e-3)
    assert out[0]["score"] == pytest.approx(0.7333, abs=1e-4)


def test_low_iou_stays_separate_and_output_is_sorted():
    # A = [0,0,10,10], B = [5,0,15,10]: IoU 50/150 = 0.33 < 0.55
    out = fuse([det(0, 0, 10, 10, 0.6)], [det(5, 0, 10, 10, 0.8)])
    assert [round(d["score"], 4) for d in out] == [0.4, 0.3]
    assert out[0]["bbox"] == pytest.approx([5, 0, 10, 10], abs=1e-3)


def test_classes_never_merge():
    out = fuse([det(10, 10, 20, 10, 0.8, c=1)], [det(10, 10, 20, 10, 0.6, c=2)])
    assert sorted((d["category_id"], round(d["score"], 4)) for d in out) == [(1, 0.4), (2, 0.3)]


def test_outside_box_is_clipped_and_empty_box_dropped_without_warnings():
    with warnings.catch_warnings():
        warnings.simplefilter("error")  # kırpma bizde: kütüphane uyarı basmamalı
        out = fuse([det(-5, 0, 10, 10, 0.8), det(-30, 0, 10, 10, 0.9)], [])
    assert len(out) == 1
    assert out[0]["bbox"] == pytest.approx([0, 0, 5, 10], abs=1e-3)


def test_skip_threshold_and_zero_score():
    assert fuse([det(10, 10, 20, 10, 0.0005)], [], skip=0.001) == []
    assert fuse([det(10, 10, 20, 10, 0.0)], []) == []  # 0 skor: 0'a bölme / NaN yok


def test_no_boxes():
    assert fuse([], []) == []


def test_bad_weights():
    with pytest.raises(ValueError, match="ağırlık"):
        fuse_image([[], []], [1.0], width=W, height=H, iou_thr=0.5)
    with pytest.raises(ValueError, match="ağırlık"):
        fuse([det(10, 10, 20, 10, 0.8)], [], (1, 0))


def test_types_and_input_untouched():
    a = [det(10, 10, 20, 10, 0.8)]
    before = [dict(d, bbox=list(d["bbox"])) for d in a]
    out = fuse(a, [])
    assert a == before
    d = out[0]
    assert set(d) == {"category_id", "bbox", "score"}
    assert type(d["category_id"]) is int and type(d["score"]) is float
    assert all(type(v) is float for v in d["bbox"])
```

```python
"""M3OT ham verisinin gerçek yapısı (H4 KAPI M). Veri toplama scripti; test yazılmaz.

Kullanım: python inspect_m3ot.py <ham_kök> <çıktı.json> [--figs KLASÖR]
Her `gt/gt.txt` için dizi özeti, ağacın ilk 3 seviyesi ve ilk iki dizinin ortadaki karesine
süzülmüş kutuların çizimi (gözle hiza kontrolü).
"""

from __future__ import annotations

import argparse
import json
from collections import Counter
from pathlib import Path

import cv2

from baykus.io import imread_any, imwrite_any

IMG = {".jpg", ".jpeg", ".png", ".bmp", ".tif", ".tiff"}


def tree(root: Path, depth: int = 3, limit: int = 12) -> list[str]:
    lines = []

    def walk(d: Path, level: int) -> None:
        kids = sorted(d.iterdir())
        dirs = [k for k in kids if k.is_dir()]
        files = [k for k in kids if k.is_file()]
        exts = Counter(f.suffix.lower() for f in files)
        lines.append(f"{'  ' * level}{d.name}/  ({len(dirs)} klasör, {dict(exts)})")
        if level + 1 < depth:
            for k in dirs[:limit]:
                walk(k, level + 1)
            if len(dirs) > limit:
                lines.append(f"{'  ' * (level + 1)}… {len(dirs) - limit} klasör daha")

    walk(root, 0)
    return lines


def gt_summary(path: Path) -> dict:
    rows = [ln.split(",") for ln in path.read_text(encoding="utf-8").splitlines() if ln.strip()]

    def col(i: int) -> dict:
        return dict(Counter(r[i].strip() for r in rows if len(r) > i).most_common(8))

    vis = [float(r[8]) for r in rows if len(r) > 8]
    frames = [int(float(r[0])) for r in rows]
    return {
        "rows": len(rows),
        "ncols": dict(Counter(len(r) for r in rows)),
        "frames": [min(frames), max(frames)] if frames else None,
        "tracks": len({r[1] for r in rows}),
        "col7_flag": col(6),
        "col8_class": col(7),
        "col9_vis": {"min": min(vis), "max": max(vis), "le_0.25": sum(v <= 0.25 for v in vis)}
        if vis
        else None,
        "head": [",".join(r) for r in rows[:3]],
    }


def seq_summary(seq: Path, raw: Path) -> dict:
    imgs = sorted(p for p in (seq / "img1").iterdir() if p.suffix.lower() in IMG)
    first = imread_any(imgs[0], cv2.IMREAD_UNCHANGED) if imgs else None
    ini = seq / "seqinfo.ini"
    return {
        "path": seq.relative_to(raw).as_posix(),
        "n_images": len(imgs),
        "exts": dict(Counter(p.suffix.lower() for p in imgs)),
        "first": imgs[0].name if imgs else None,
        "last": imgs[-1].name if imgs else None,
        "shape": list(first.shape) if first is not None else None,
        "dtype": str(first.dtype) if first is not None else None,
        "seqinfo": ini.read_text(encoding="utf-8") if ini.exists() else None,
        "gt": gt_summary(seq / "gt" / "gt.txt"),
    }


def draw(seq: Path, out: Path) -> None:
    imgs = sorted(p for p in (seq / "img1").iterdir() if p.suffix.lower() in IMG)
    mid = imgs[len(imgs) // 2]
    frame = int("".join(ch for ch in mid.stem if ch.isdigit()) or 0)
    img = imread_any(mid, cv2.IMREAD_COLOR)
    for ln in (seq / "gt" / "gt.txt").read_text(encoding="utf-8").splitlines():
        f = ln.split(",")
        if len(f) < 9 or int(float(f[0])) != frame:
            continue
        x, y, w, h = (int(float(v)) for v in f[2:6])
        ok = int(float(f[6])) == 1 and int(float(f[7])) == 1 and float(f[8]) > 0.25
        cv2.rectangle(img, (x, y), (x + w, y + h), (0, 255, 0) if ok else (0, 0, 255), 1)
    name = seq.relative_to(seq.parents[2]).as_posix().replace("/", "__")
    imwrite_any(out / f"{name}__{mid.stem}.jpg", img)


def main() -> None:
    p = argparse.ArgumentParser()
    p.add_argument("raw", type=Path)
    p.add_argument("out", type=Path)
    p.add_argument("--figs", type=Path, default=None)
    a = p.parse_args()
    seqs = sorted(g.parent.parent for g in a.raw.rglob("gt/gt.txt"))
    report = {
        "root": str(a.raw),
        "tree": tree(a.raw),
        "sequences": [seq_summary(s, a.raw) for s in seqs],
        "other_files": sorted(
            f"{f.relative_to(a.raw).as_posix()} ({f.stat().st_size} B)"
            for f in a.raw.rglob("*")
            if f.is_file() and f.suffix.lower() in {".json", ".txt", ".ini", ".csv", ".md"}
            and f.name != "gt.txt"
        )[:40],
    }
    if a.figs:
        a.figs.mkdir(parents=True, exist_ok=True)
        for s in seqs[:4]:
            draw(s, a.figs)
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(report, indent=2, ensure_ascii=False), encoding="utf-8")
    print("\n".join(report["tree"]))
    print(f"\n{len(seqs)} dizi")
    for s in report["sequences"]:
        g = s["gt"]
        print(f"{s['path']} | {s['n_images']} görüntü {s['exts']} {s['first']}..{s['last']} "
              f"{s['shape']} | gt {g['rows']} satır, kare {g['frames']}, iz {g['tracks']}, "
              f"flag {g['col7_flag']}, sınıf {g['col8_class']}, vis {g['col9_vis']}")  # fmt: skip
    print("\ndiğer dosyalar:", *report["other_files"], sep="\n  ")


if __name__ == "__main__":
    main()
```

```python
def make_raw(root: Path, layout: str, extra_rgb_only_seq: bool = False) -> Path:
    """seqA: 3 ortak kare + sadece RGB'de 4. kare; seqB: 2 ortak kare. 40×32.
```

```python
"""M3OT → bizim işlenmiş düzen (FR-D8, kanca K5, spec §8.3).

Ham düzen (Spike C, KAPI M): her dizi img1/*.jpg ve gt/gt.txt taşıyan bir klasör. gt.txt (MOT):
frame, track_id, x, y, w, h, flag, class, visibility. Bölüm ve modalite yol parçalarından ya da
dizi adının eklerinden okunur (test/RGB/seq03, test/seq03_ir, test/seq03/ir).
Çıktı DroneVehicle düzeniyle aynı; stem = <dizi>_<kare:06d>. Dizi, kare ve iz numarası
meta/m3ot_tracks.csv'de korunur (K5: Faz 3 takip, Faz 4 hareketli/duran).
Süzgeç resmi tools/convert_M3OT_to_yolo.py ile aynı: flag == 1, sınıf == 1, görünürlük > 0.25.
"""

from __future__ import annotations

import csv
import json
import multiprocessing
import re
import shutil
from collections import defaultdict
from collections.abc import Iterable
from concurrent.futures import ProcessPoolExecutor
from dataclasses import astuple, dataclass
from pathlib import Path

import cv2

from baykus.data.coco import yolo_to_coco
from baykus.data.convert import to_single_channel
from baykus.data.yamls import write_data_yamls
from baykus.io import imread_any, imwrite_any

CLASS_NAME = "vehicle"
SPLITS = ("train", "val", "test")
MODALITY_TOKENS = {"rgb": {"rgb", "visible", "vis"}, "ir": {"ir", "infrared", "thermal", "tir"}}
MIN_VISIBILITY = 0.25
IMG_EXTS = {".jpg", ".jpeg", ".png"}
_SEP = re.compile(r"[_\-. ]+")
TRACK_FIELDS = ["split", "modality", "seq", "frame", "stem", "track_id", "x", "y", "w", "h",
                "flag", "cls", "vis"]  # fmt: skip


@dataclass(frozen=True)
class Seq:
    split: str
    name: str
    modality: str
    path: Path


@dataclass(frozen=True)
class GtRow:
    frame: int
    track_id: int
    x: float
    y: float
    w: float
    h: float
    flag: int
    cls: int
    vis: float


def _tokens(part: str) -> list[str]:
    return [t for t in _SEP.split(part) if t]


def _modality(part: str) -> str | None:
    toks = {t.lower() for t in _tokens(part)}
    found = [m for m, names in MODALITY_TOKENS.items() if toks & names]
    if len(found) > 1:
        raise ValueError(f"{part!r}: hem RGB hem IR adı geçiyor")
    return found[0] if found else None


def _strip(part: str) -> str:
    names = set().union(*MODALITY_TOKENS.values())
    return "_".join(t for t in _tokens(part) if t.lower() not in names)


def classify(rel: Path) -> tuple[str, str, str]:
    """Ham köke göre dizi klasörü → (bölüm, modalite, dizi adı)."""
    parts = Path(rel).parts
    split = next((p.lower() for p in parts if p.lower() in SPLITS), None)
    mods = {m for p in parts if (m := _modality(p))}
    if split is None or len(mods) != 1:
        raise ValueError(
            f"{Path(rel).as_posix()}: bölüm ya da modalite yoldan okunamadı "
            "(KAPI M çıktısını Claude'a getir)"
        )
    name = _strip(parts[-1]) or _strip(parts[-2])  # test/seq03/ir → seq03
    return split, mods.pop(), name


def find_sequences(raw: Path) -> list[Seq]:
    raw = Path(raw)
    out = []
    for gt in sorted(raw.rglob("gt/gt.txt")):
        seq_dir = gt.parent.parent
        if (seq_dir / "img1").is_dir():
            split, modality, name = classify(seq_dir.relative_to(raw))
            out.append(Seq(split, name, modality, seq_dir))
    return out


def pair_sequences(seqs: Iterable[Seq]) -> tuple[list[tuple[Seq, Seq]], list[Seq]]:
    by: dict[tuple[str, str], dict[str, Seq]] = defaultdict(dict)
    for s in seqs:
        if s.modality in by[(s.split, s.name)]:
            raise ValueError(f"{s.split}/{s.name}: iki {s.modality} dizisi var")
        by[(s.split, s.name)][s.modality] = s
    pairs, alone = [], []
    for key in sorted(by):
        d = by[key]
        if "rgb" in d and "ir" in d:
            pairs.append((d["rgb"], d["ir"]))
        else:
            alone += [d[m] for m in sorted(d)]
    return pairs, alone


def read_gt(path: Path) -> list[GtRow]:
    rows = []
    for n, line in enumerate(Path(path).read_text(encoding="utf-8").splitlines(), 1):
        if not line.strip():
            continue
        f = [v.strip() for v in line.split(",")]
        if len(f) < 9:
            raise ValueError(f"{path}:{n}: 9 sütun bekleniyordu, {len(f)} var")
        i = [int(float(v)) for v in (f[0], f[1], f[6], f[7])]
        x, y, w, h, vis = (float(v) for v in (f[2], f[3], f[4], f[5], f[8]))
        rows.append(GtRow(i[0], i[1], x, y, w, h, i[2], i[3], vis))
    return rows


def keep(r: GtRow) -> bool:
    return r.flag == 1 and r.cls == 1 and r.vis > MIN_VISIBILITY and r.w > 0 and r.h > 0


def yolo_lines(rows: Iterable[GtRow], width: int, height: int) -> list[str]:
    """Kutu görüntüye kırpılır, boş kalan atlanır. Sınıf 0: değerlendirme sınıfsız (single_cls)."""
    out = []
    for r in rows:
        x1, y1 = max(r.x, 0.0), max(r.y, 0.0)
        x2, y2 = min(r.x + r.w, float(width)), min(r.y + r.h, float(height))
        if x2 <= x1 or y2 <= y1:
            continue
        cx, cy = (x1 + x2) / 2 / width, (y1 + y2) / 2 / height
        out.append(f"0 {cx:.6f} {cy:.6f} {(x2 - x1) / width:.6f} {(y2 - y1) / height:.6f}")
    return out


def frame_images(img_dir: Path) -> dict[int, Path]:
    out = {}
    for p in sorted(Path(img_dir).iterdir()):
        if p.suffix.lower() not in IMG_EXTS:
            continue
        digits = re.findall(r"\d+", p.stem)
        if not digits:
            raise ValueError(f"{p}: kare numarası okunamadı")
        out[int(digits[-1])] = p
    return out


def _write_frame(job: tuple) -> tuple[int, int]:
    rgb_src, ir_src, out, split, stem = job
    bgr = imread_any(rgb_src, cv2.IMREAD_COLOR)
    if bgr is None:
        raise OSError(f"okunamadı: {rgb_src}")
    ir = imread_any(ir_src, cv2.IMREAD_UNCHANGED)
    if ir is None:
        raise OSError(f"okunamadı: {ir_src}")
    ir = to_single_channel(ir)
    if bgr.shape[:2] != ir.shape[:2]:
        raise ValueError(f"{stem}: RGB {bgr.shape[:2]} ≠ IR {ir.shape[:2]}")
    dst = out / "rgb/images" / split / f"{stem}.jpg"
    dst.parent.mkdir(parents=True, exist_ok=True)
    if rgb_src.suffix.lower() in (".jpg", ".jpeg"):
        shutil.copy2(rgb_src, dst)  # yeniden sıkıştırma yok
    else:
        imwrite_any(dst, bgr)
    imwrite_any(out / "ir/images" / split / f"{stem}.png", ir)
    return ir.shape[1], ir.shape[0]


def _write_csv(path: Path, header: list[str], rows: list[list]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", newline="", encoding="utf-8") as f:
        w = csv.writer(f)
        w.writerow(header)
        w.writerows(rows)


def convert_m3ot(
    raw: Path, out: Path, splits: Iterable[str] = ("test",), workers: int = 1
) -> dict:
    raw, out, splits = Path(raw), Path(out), tuple(splits)
    pairs, alone = pair_sequences(s for s in find_sequences(raw) if s.split in splits)
    if not pairs:
        raise ValueError(f"{raw}: {list(splits)} için RGB-IR dizi çifti bulunamadı")
    stats = {
        "pairs": 0,
        "sequences": len(pairs),
        "unpaired_sequences": [f"{s.split}/{s.name}/{s.modality}" for s in alone],
        "frames_rgb_only": 0,
        "frames_ir_only": 0,
        "boxes": {"rgb": 0, "ir": 0},
        "dropped_rows": {"rgb": 0, "ir": 0},
    }
    jobs, labels, index, tracks = [], {}, [], []
    for rgb, ir in pairs:
        f_rgb, f_ir = frame_images(rgb.path / "img1"), frame_images(ir.path / "img1")
        common = sorted(set(f_rgb) & set(f_ir))
        stats["frames_rgb_only"] += len(set(f_rgb) - set(f_ir))
        stats["frames_ir_only"] += len(set(f_ir) - set(f_rgb))
        by_frame: dict[str, dict[int, list[GtRow]]] = {}
        for m, seq in (("rgb", rgb), ("ir", ir)):
            by_frame[m] = defaultdict(list)
            for r in read_gt(seq.path / "gt" / "gt.txt"):
                stem = f"{rgb.name}_{r.frame:06d}"
                tracks.append([rgb.split, m, rgb.name, r.frame, stem, *astuple(r)[1:]])
                if keep(r):
                    by_frame[m][r.frame].append(r)
                else:
                    stats["dropped_rows"][m] += 1
        for fr in common:
            stem = f"{rgb.name}_{fr:06d}"
            jobs.append((f_rgb[fr], f_ir[fr], out, rgb.split, stem))
            labels[(rgb.split, stem)] = {m: by_frame[m].get(fr, []) for m in ("rgb", "ir")}
            index.append([rgb.split, stem, rgb.name, fr])
    if workers <= 1:
        sizes = [_write_frame(j) for j in jobs]
    else:
        ctx = multiprocessing.get_context("spawn")
        with ProcessPoolExecutor(max_workers=workers, mp_context=ctx) as ex:
            sizes = list(ex.map(_write_frame, jobs, chunksize=32))
    if len(set(sizes)) != 1:
        raise ValueError(f"görüntü boyutları farklı: {sorted(set(sizes))[:5]}")
    width, height = sizes[0]
    for (split, stem), per_mod in labels.items():
        for m, rows in per_mod.items():
            lines = yolo_lines(rows, width, height)
            stats["boxes"][m] += len(lines)
            p = out / f"labels_{m}" / split / f"{stem}.txt"
            p.parent.mkdir(parents=True, exist_ok=True)
            p.write_text("".join(ln + "\n" for ln in lines), encoding="utf-8")
    (out / "coco").mkdir(parents=True, exist_ok=True)
    for split in splits:
        stems = sorted(stem for s, stem in labels if s == split)
        for m in ("ir", "rgb"):
            if stems:
                coco = yolo_to_coco(
                    out / f"labels_{m}" / split, stems, width, height, single_class=CLASS_NAME
                )
                (out / "coco" / f"{split}_{m}.json").write_text(
                    json.dumps(coco), encoding="utf-8"
                )
    stats["pairs"] = len(jobs)
    meta = out / "meta"
    _write_csv(meta / "m3ot_index.csv", ["split", "stem", "seq", "frame"], index)
    _write_csv(meta / "m3ot_tracks.csv", TRACK_FIELDS, tracks)
    (meta / "m3ot_stats.json").write_text(
        json.dumps(stats, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    write_data_yamls(out)
    return stats
```

```python
"""Geç füzyon (FR-V7, kanca K4): iki modelin COCO tahminleri → görüntü görüntü WBF.

Ağırlıklar görüntü başına dışarıdan verilir: {image_id: (w_rgb, w_ir)} ya da bütün görüntüler
için tek (w_rgb, w_ir). E4 (sabit) ve E4b (aydınlatma dilimine göre) bunun özel hâlleri; Faz 2'nin
kare başına uyarlanan ağırlığı da aynı arayüzü kullanır.
"""

from __future__ import annotations

import csv
import json
import multiprocessing
from collections import defaultdict
from collections.abc import Iterable, Mapping
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

from baykus.data.illumination import assign
from baykus.fusion.wbf import fuse_image
from baykus.robustness.corruptions import Condition

Weights = tuple[float, float] | Mapping[int, tuple[float, float]]
CHUNK = 64


def _group(dets: Iterable[dict]) -> dict[int, list[dict]]:
    out: dict[int, list[dict]] = defaultdict(list)
    for x in dets:
        out[x["image_id"]].append(x)
    return out


def _fuse_chunk(job: tuple) -> list[dict]:
    items, iou_thr, skip_box_thr = job
    out = []
    for img_id, (width, height), wts, a, b in items:
        fused = fuse_image(
            [a, b], wts, width=width, height=height, iou_thr=iou_thr, skip_box_thr=skip_box_thr
        )
        out += [{"image_id": img_id, **x} for x in fused]
    return out


def fuse_predictions(
    rgb: list[dict],
    ir: list[dict],
    images: Mapping[int, tuple[int, int]],
    weights: Weights,
    *,
    iou_thr: float,
    skip_box_thr: float = 0.001,
    only: Iterable[int] | None = None,
    workers: int = 1,
) -> list[dict]:
    by_rgb, by_ir = _group(rgb), _group(ir)
    unknown = sorted((set(by_rgb) | set(by_ir)) - set(images))
    if unknown:
        raise ValueError(
            f"GT'de olmayan image_id: {unknown[:5]} (tahmin dosyası başka bölümün mü?)"
        )
    ids = sorted(images) if only is None else sorted(set(only))
    missing = [i for i in ids if i not in images]
    if missing:
        raise ValueError(f"GT'de olmayan image_id: {missing[:5]}")
    items = []
    for i in ids:
        a, b = by_rgb.get(i, []), by_ir.get(i, [])
        if not a and not b:
            continue
        if isinstance(weights, Mapping):
            if i not in weights:
                raise KeyError(f"image_id {i} için ağırlık yok")
            wts = weights[i]
        else:
            wts = weights
        items.append((i, images[i], tuple(wts), a, b))
    jobs = [(items[k : k + CHUNK], iou_thr, skip_box_thr) for k in range(0, len(items), CHUNK)]
    if workers <= 1:
        parts = [_fuse_chunk(j) for j in jobs]
    else:
        ctx = multiprocessing.get_context("spawn")
        with ProcessPoolExecutor(max_workers=workers, mp_context=ctx) as ex:
            parts = list(ex.map(_fuse_chunk, jobs))
    return [x for p in parts for x in p]


def levels_from_slices(slices: Mapping[str, Iterable[int]]) -> dict[int, str]:
    out: dict[int, str] = {}
    for level, ids in slices.items():
        for i in ids:
            if i in out:
                raise ValueError(f"image_id {i} iki aydınlatma diliminde: {out[i]}, {level}")
            out[i] = level
    return out


def illumination_weights(
    levels: Mapping[int, str], ir_weight: Mapping[str, float], w_rgb: float = 1.0
) -> dict[int, tuple[float, float]]:
    missing = sorted(set(levels.values()) - set(ir_weight))
    if missing:
        raise ValueError(f"şu aydınlatma dilimleri için IR ağırlığı yok: {missing}")
    return {i: (w_rgb, float(ir_weight[lv])) for i, lv in levels.items()}


def condition_levels(
    brightness: Mapping[int, float], thresholds: tuple[float, float], cond: Condition
) -> dict[int, str]:
    """Bozulmuş RGB'nin aydınlatma etiketi (FR-V7: çıkarımda RGB'den türetilir).

    Parlaklık (HSV V ortalaması): karartmada çarpanla ölçeklenir (yuvarlama ±0,5), RGB kaybında
    0; bulanıklık, gürültü ve hizasızlıkta değişmez kabul edilir (Görev 8 / Spec yorumları 5).
    """
    if cond.corruption == "darken":
        f = cond.severity
    elif cond.corruption == "drop_rgb":
        f = 0.0
    else:
        f = 1.0
    ids = sorted(brightness)
    labels = assign(np.array([brightness[i] * f for i in ids]), thresholds)
    return dict(zip(ids, labels, strict=True))


def load_brightness(
    meta: Path, split: str, gt_images: list[dict]
) -> tuple[dict[int, float], tuple[float, float]]:
    meta = Path(meta)
    by_stem = {}
    with (meta / "illumination.csv").open(encoding="utf-8") as f:
        for row in csv.DictReader(f):
            if row["split"] == split:
                by_stem[row["stem"]] = float(row["brightness"])
    out = {}
    for im in gt_images:
        stem = Path(im["file_name"]).stem
        if stem not in by_stem:
            raise ValueError(f"{stem}: illumination.csv'de {split} parlaklığı yok")
        out[im["id"]] = by_stem[stem]
    th = json.loads((meta / "illumination_thresholds.json").read_text(encoding="utf-8"))
    return out, (float(th["t_dark"]), float(th["t_day"]))
```

```python
"""Geç füzyon config'i (FR-V7). Eğitim config'inden ayrı: model eğitilmez, iki modelin
tahminleri birleşir. Bilinmeyen anahtar ConfigError (eğitim config'iyle aynı kural)."""

from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

import yaml

from baykus.train.config import LABEL_SETS, ConfigError, _check_keys, config_hash

_TOP = {"experiment", "mode", "model", "rgb", "ir", "wbf", "eval"}
_WBF = {"iou_thr", "ir_weight", "skip_box_thr", "per_illumination"}
_EVAL = {"label_sets", "n_boot", "seed"}


@dataclass(frozen=True)
class FusionConfig:
    experiment: str
    model: str
    rgb: str
    ir: str
    iou_thr: tuple[float, ...]
    ir_weight: tuple[float, ...]
    skip_box_thr: float = 0.001
    per_illumination: bool = False
    label_sets: tuple[str, ...] = LABEL_SETS
    n_boot: int = 1000
    seed: int = 42
    config_hash: str = ""
    mode: str = "late_fusion"


def parse_fusion_config(raw: dict) -> FusionConfig:
    _check_keys(raw, _TOP, "config")
    for key in ("experiment", "mode", "model", "rgb", "ir", "wbf"):
        if key not in raw:
            raise ConfigError(f"config: zorunlu anahtar eksik: {key}")
    if raw["mode"] != "late_fusion":
        raise ConfigError(f"mode: geç füzyon config'inde 'late_fusion' olmalı, {raw['mode']!r}")
    wbf = _check_keys(raw["wbf"], _WBF, "wbf")
    iou = tuple(float(x) for x in wbf.get("iou_thr") or ())
    irw = tuple(float(x) for x in wbf.get("ir_weight") or ())
    per_ill = bool(wbf.get("per_illumination", False))
    if not irw or any(w <= 0 for w in irw):
        raise ConfigError(f"wbf.ir_weight: boş olmayan pozitif liste olmalı, {irw}")
    if any(not 0 < x < 1 for x in iou):
        raise ConfigError(f"wbf.iou_thr: değerler 0-1 arası olmalı, {iou}")
    if per_ill and iou:
        raise ConfigError("wbf: per_illumination iken iou_thr verilmez; E4'ün seçtiği kullanılır")
    if not per_ill and not iou:
        raise ConfigError("wbf.iou_thr boş olamaz (sadece E4b'de E4'ten gelir)")
    skip = float(wbf.get("skip_box_thr", 0.001))
    if not 0 <= skip < 1:
        raise ConfigError(f"wbf.skip_box_thr: 0-1 arası olmalı, {skip}")
    ev = _check_keys(raw.get("eval") or {}, _EVAL, "eval")
    label_sets = tuple(ev.get("label_sets", LABEL_SETS))
    bad = [x for x in label_sets if x not in LABEL_SETS]
    if bad:
        raise ConfigError(f"eval.label_sets: geçersiz {bad}; seçenekler: {LABEL_SETS}")
    return FusionConfig(
        experiment=str(raw["experiment"]),
        model=str(raw["model"]),
        rgb=str(raw["rgb"]),
        ir=str(raw["ir"]),
        iou_thr=iou,
        ir_weight=irw,
        skip_box_thr=skip,
        per_illumination=per_ill,
        label_sets=label_sets,
        n_boot=int(ev.get("n_boot", 1000)),
        seed=int(ev.get("seed", 42)),
        config_hash=config_hash(raw),
    )


def load_fusion_config(path: Path) -> FusionConfig:
    raw = yaml.safe_load(Path(path).read_text(encoding="utf-8"))
    if not isinstance(raw, dict):
        raise ConfigError(f"{path}: config bir sözlük (YAML mapping) olmalı")
    return parse_fusion_config(raw)
```

```python
"""E4 / E4b parametre seçimi (FR-V7): sadece val, IR etiketleri, ölçüt mAP50.
Eşitlikte ızgara sırasındaki ilk hücre (belirlenimci)."""

from __future__ import annotations

from collections.abc import Iterable, Mapping

from pycocotools.coco import COCO

from baykus.eval.coco_eval import evaluate_coco
from baykus.fusion.config import FusionConfig
from baykus.fusion.late import fuse_predictions


def grid_search(
    rgb: list[dict],
    ir: list[dict],
    gt: COCO,
    *,
    iou_thrs: Iterable[float],
    ir_weights: Iterable[float],
    skip_box_thr: float,
    img_ids: list[int] | None = None,
    workers: int = 1,
) -> list[dict]:
    images = {im["id"]: (im["width"], im["height"]) for im in gt.dataset["images"]}
    rows = []
    for iou in iou_thrs:
        for w in ir_weights:
            fused = fuse_predictions(rgb, ir, images, (1.0, float(w)), iou_thr=iou,
                                     skip_box_thr=skip_box_thr, only=img_ids,
                                     workers=workers)  # fmt: skip
            rows.append(
                {"iou_thr": iou, "ir_weight": float(w),
                 "map50": evaluate_coco(gt, fused, img_ids)["map50"]}  # fmt: skip
            )
            print(f"iou {iou} · ir {w}: map50 {rows[-1]['map50']:.4f}", flush=True)
    return rows


def best(rows: list[dict]) -> dict:
    if not rows:
        raise ValueError("boş ızgara")
    return max(rows, key=lambda r: r["map50"])  # max eşitlikte ilkini döner


def select_constant(rgb, ir, gt: COCO, cfg: FusionConfig, workers: int = 1) -> dict:
    grid = grid_search(rgb, ir, gt, iou_thrs=cfg.iou_thr, ir_weights=cfg.ir_weight,
                       skip_box_thr=cfg.skip_box_thr, workers=workers)  # fmt: skip
    b = best(grid)
    return {"grid": grid, "selected": {"iou_thr": b["iou_thr"], "w_rgb": 1.0,
                                       "ir_weight": b["ir_weight"]}}  # fmt: skip


def select_per_illumination(
    rgb, ir, gt: COCO, cfg: FusionConfig, slices: Mapping[str, list[int]], iou_thr: float,
    workers: int = 1,
) -> dict:  # fmt: skip
    grid, chosen = [], {}
    for level, ids in slices.items():
        if not ids:
            continue
        rows = grid_search(rgb, ir, gt, iou_thrs=[iou_thr], ir_weights=cfg.ir_weight,
                           skip_box_thr=cfg.skip_box_thr, img_ids=list(ids),
                           workers=workers)  # fmt: skip
        grid += [{**r, "slice": level} for r in rows]
        chosen[level] = best(rows)["ir_weight"]
    if not chosen:
        raise ValueError("aydınlatma dilimi yok")
    return {"grid": grid, "selected": {"iou_thr": iou_thr, "w_rgb": 1.0, "ir_weight": chosen}}
```

```python
"""baykus-fusion: geç füzyon (FR-V7; E4, E4b; kanca K4). GPU gerekmez.

select — val'de WBF parametrelerini seçer; seçim JSON'u + birleşik tahminlerin run JSON'ları.

import json
import sys

from pathlib import Path

SEL_DIR = "fusion"
LABEL_SET = "ir"  # seçim ölçütü ana etiket setinde (ADR 0003)


class FusionError(ValueError):
    pass


def selection_path(results: Path, experiment: str, split: str) -> Path:
    return Path(results) / SEL_DIR / f"{experiment}_{split}_selection.json"


def _weights(sel: dict, levels: dict[int, str] | None):
    s = sel["selected"]
    if isinstance(s["ir_weight"], dict):
        if levels is None:
            raise FusionError("aydınlatmaya göre ağırlık için görüntülerin aydınlatma etiketi gerekli")
        return illumination_weights(levels, s["ir_weight"], s["w_rgb"])
    return (s["w_rgb"], s["ir_weight"])


def fuse_and_record(
    sel: dict,
    cfg: FusionConfig,
    *,
    rgb_preds: Path,
    ir_preds: Path,
    root: Path,
    meta: Path,
    split: str,
    project: Path,
    results: Path,
    n_boot: int,
    workers: int = 1,
    dataset: str = DEFAULT_DATASET,
    class_agnostic: bool = False,
    condition: dict | None = None,
    levels: dict[int, str] | None = None,
    label_sets: tuple[str, ...] | None = None,
    results_subdir: str = "runs",
) -> list[Path]:
    label_sets = label_sets or cfg.label_sets
    name = run_name(sel["run_id"], split, dataset, condition)
    outs = [Path(results) / results_subdir / f"{name}_{ls}.json" for ls in label_sets]
    if all(p.exists() for p in outs):
        print(f"zaten var, atlandı: {name}")
        return outs
    root = Path(root)
    gt_ir = load_coco(root / "coco" / f"{split}_ir.json")
    images = {im["id"]: (im["width"], im["height"]) for im in gt_ir.dataset["images"]}
    weights = _weights(sel, levels)
    rgb, ir = load_predictions(Path(rgb_preds)), load_predictions(Path(ir_preds))
    if class_agnostic:
        rgb = [{**d, "category_id": 1} for d in rgb]
        ir = [{**d, "category_id": 1} for d in ir]
    s = sel["selected"]
    fused = fuse_predictions(rgb, ir, images, weights, iou_thr=s["iou_thr"],
                             skip_box_thr=sel["skip_box_thr"], workers=workers)  # fmt: skip
    save_predictions(Path(project) / name / "predictions_coco.json.gz", fused)
    info = {
        "rgb_run_id": sel["rgb_run_id"],
        "ir_run_id": sel["ir_run_id"],
        "rgb_preds": str(rgb_preds),
        "ir_preds": str(ir_preds),
        "iou_thr": s["iou_thr"],
        "w_rgb": s["w_rgb"],
        "ir_weight": s["ir_weight"],
        "skip_box_thr": sel["skip_box_thr"],
        "conf_type": "avg",
        "selected_on": sel["split"],
        "selection": sel["experiment"],
    }
    git = git_info()
    for ls, out in zip(label_sets, outs, strict=True):
        gt = load_coco(root / "coco" / f"{split}_{ls}.json")
        metrics = build_metrics(gt, fused, load_slices(meta, split), n_boot=n_boot, seed=cfg.seed)
        rec = eval_record(
            run_id=sel["run_id"],
            config=sel["config"],
            config_hash=sel["config_hash"],
            git=git,
            seed=cfg.seed,
            dataset=dataset,
            split=split,
            label_set=ls,
            model=model_block(cfg.model, "late_fusion"),
            weights=None,
            n_boot=n_boot,
            dropped=0,
            metrics=metrics,
            class_agnostic=class_agnostic,
        )
        rec["fusion"] = info
        if condition is not None:
            rec["robustness"] = condition
        write_json(out, rec)
        print(out, f"map50={metrics['map50']:.4f}", flush=True)
    return outs


def _cmd_select(a: argparse.Namespace) -> int:
    cfg = load_fusion_config(Path(a.config))
    root, meta = Path(a.data_root), Path(a.meta)
    out = selection_path(a.results, cfg.experiment, "val")
    if out.exists():
        sel = json.loads(out.read_text(encoding="utf-8"))
        print(f"seçim zaten var, yeniden aranmadı: {out}")
    else:
        gt = load_coco(root / "coco" / f"val_{LABEL_SET}.json")
        rgb, ir = load_predictions(Path(a.rgb_preds)), load_predictions(Path(a.ir_preds))
        t0 = time.monotonic()
        if cfg.per_illumination:
            if not a.base:
                raise FusionError("per_illumination: --base (E4 seçim JSON'u) gerekli")
            base = json.loads(Path(a.base).read_text(encoding="utf-8"))
            if (base["rgb_run_id"], base["ir_run_id"]) != (a.rgb_run_id, a.ir_run_id):
                raise FusionError("--base başka modellerle seçilmiş; aynı E1/E2 olmalı")
            slices = load_slices(meta, "val")
            if not slices:
                raise FusionError(f"{meta}/slices_val.json yok")
            found = select_per_illumination(rgb, ir, gt, cfg, slices,
                                            base["selected"]["iou_thr"], a.workers)  # fmt: skip
        else:
            found = select_constant(rgb, ir, gt, cfg, a.workers)
        git = git_info()
        sel = {
            "experiment": cfg.experiment,
            "config": a.config,
            "config_hash": cfg.config_hash,
            "run_id": make_run_id(cfg.experiment, cfg.model, git["git_commit"], utc_now()),
            "split": "val",
            "label_set": LABEL_SET,
            "criterion": "map50",
            "rgb_run_id": a.rgb_run_id,
            "ir_run_id": a.ir_run_id,
            "rgb_preds": str(a.rgb_preds),
            "ir_preds": str(a.ir_preds),
            "skip_box_thr": cfg.skip_box_thr,
            **found,
            "seconds": round(time.monotonic() - t0, 1),
            "created_at": iso(utc_now()),
        }
        write_json(out, sel)
        print(out, json.dumps(sel["selected"], ensure_ascii=False), f"{sel['seconds']} sn")
    levels = levels_from_slices(load_slices(meta, "val") or {}) if cfg.per_illumination else None
    fuse_and_record(sel, cfg, rgb_preds=Path(a.rgb_preds), ir_preds=Path(a.ir_preds), root=root,
                    meta=meta, split="val", project=Path(a.project), results=Path(a.results),
                    n_boot=a.n_boot or cfg.n_boot, workers=a.workers,
                    levels=levels)  # fmt: skip
    return 0

def main(argv: list[str] | None = None) -> int:
```

```python
def compose_sources(cond: Condition) -> tuple[str, str]:
    """Geç füzyonun koşul girdileri: bozulma bir modaliteye dokunmuyorsa o modelin temiz
    tahmini. Hizasızlık IR'yi kaydırır: E2'nin kaydırılmış IR tahmini (--ir-misalign)."""
    rgb = cond.id if applies(cond, "rgb") else "clean"
    ir = cond.id if applies(cond, "ir") or cond.corruption == "misalign" else "clean"
    return rgb, ir


def _load_selection(a: argparse.Namespace) -> tuple[dict, FusionConfig]:
    cfg = load_fusion_config(Path(a.config))
    sel = json.loads(Path(a.selection).read_text(encoding="utf-8"))
    if sel["experiment"] != cfg.experiment:
        raise FusionError(f"seçim {sel['experiment']!r}, config {cfg.experiment!r}: eşleşmiyor")
    return sel, cfg


def _cmd_robust(a: argparse.Namespace) -> int:
    sel, cfg = _load_selection(a)
    root, meta, rp = Path(a.data_root), Path(a.meta), Path(a.robust_preds)
    conds = [c for c in suite_conditions(a.only) if applies(c, "late_fusion")]
    srcs = {}
    for c in conds:
        cid_rgb, cid_ir = compose_sources(c)
        srcs[c.id] = (
            rp / f"{sel['rgb_run_id']}_{a.split}_{cid_rgb}" / "predictions_coco.json.gz",
            rp / f"{sel['ir_run_id']}_{a.split}_{cid_ir}" / "predictions_coco.json.gz",
            {"rgb": cid_rgb, "ir": cid_ir},
        )
    missing = sorted({str(p) for r, i, _ in srcs.values() for p in (r, i) if not p.exists()})
    if missing:
        raise FusionError(
            "tahmin dosyası yok (03_robust bitti mi? E2 hizasızlığı için --ir-misalign): "
            + ", ".join(missing)
        )
    brightness = th = None
    if cfg.per_illumination:
        gt = load_coco(root / "coco" / f"{a.split}_ir.json")
        brightness, th = load_brightness(meta, a.split, gt.dataset["images"])
    for c in conds:
        rgb_p, ir_p, sources = srcs[c.id]
        info = {"id": c.id, "corruption": c.corruption, "severity": c.severity,
                "target": c.target, "seed": a.seed, "sources": sources}  # fmt: skip
        fuse_and_record(
            sel, cfg, rgb_preds=rgb_p, ir_preds=ir_p, root=root, meta=meta, split=a.split,
            project=Path(a.project), results=Path(a.results), n_boot=a.n_boot or cfg.n_boot,
            workers=a.workers, condition=info, label_sets=(LABEL_SET,),
            results_subdir=ROBUST_DIR,
            levels=condition_levels(brightness, th, c) if cfg.per_illumination else None,
        )  # fmt: skip
    return 0


def _cmd_apply(a: argparse.Namespace) -> int:
    sel, cfg = _load_selection(a)
    if isinstance(sel["selected"]["ir_weight"], dict):
        raise FusionError(
            f"aydınlatmaya göre ağırlık (E4b) {a.dataset}'te uygulanmaz: aydınlatma etiketi yok"
        )
    root = Path(a.data_root)
    fuse_and_record(
        sel, cfg, rgb_preds=Path(a.rgb_preds), ir_preds=Path(a.ir_preds), root=root,
        meta=Path(a.meta) if a.meta else root / "meta", split=a.split, project=Path(a.project),
        results=Path(a.results), n_boot=a.n_boot or cfg.n_boot, workers=a.workers,
        dataset=a.dataset, class_agnostic=a.class_agnostic,
        results_subdir="runs" if a.dataset == DEFAULT_DATASET else a.dataset,
    )  # fmt: skip
    return 0
```
